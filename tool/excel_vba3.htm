Private Function PasteSheetHP(ByVal AURL As String _
                            , ByRef ASheet As Worksheet) As Boolean
'参照設定する場合
'□ Microsoft HTML Object Library
'と
'□ Microsoft Internet Controls
'(※↑がもし見つからない場合は、
'Microsoft Browser Helpersに
'チェックを入れてください。)
    Dim wkIE As New SHDocVw.InternetExplorer
    Dim wkDO As New MSForms.DataObject
    Dim wkDoc As MSHTML.HTMLDocument
    Dim wkText As String

    wkIE.Visible = True
    wkIE.navigate URL:=AURL
    
    ' HPが表示されるまで待つ
    Do
        DoEvents
    Loop Until (Not wkIE.Busy) And _
               (wkIE.readyState = READYSTATE_COMPLETE)
    ' ＨＰをテキストで取得する
    Set wkDoc = wkIE.document
    'wkText = wkDoc.body.innerText
    wkText = wkDoc.body.innerHTML
    ' IEを終了する
    wkIE.Quit

    wkDO.Clear
    ' DataObjectを介してクリップボードに送る
    wkDO.SetText Text:=wkText
    wkDO.PutInClipboard
    ' シートに貼り付ける
    Application.Goto Reference:=ASheet.Range("A1")
    ASheet.Paste

End Function

Sub 数字に続く全ての横線を置き換える()
    Dim c As Range
    Dim r As Range
    Dim s As String, s2 As String, s3 As String
    Set r = Selection
    For Each c In r
        s = c.Value
        For i = 1 To Len(s)
            s = c.Value
            s2 = Mid(s, i, 2)
            s3 = Right$(s2, 1)
            '数字の後に続く
            'ハイフン (四分)
            'ダッシュ (二分)
            'ダッシュ (全角)
            '長音記号
            '横細線素片
            '横太線素片
            'ダッシュ (二分)
            'ハイフンマイナス
            '機種依存文字の一つを"－（マイナス）"に
            '置き換えます。
            'ハイフンに置き換えるべきところですが、
            'エクセルの使用としてまま日付に変換されてしまうので
            'ハイフンを全角変換したときに"－（マイナス）"と
            '同じ文字に置き換えられるので
            'マイナス記号と置き換える事にしました。
            If IsNumeric(Left$(s2, 1)) _
                And (s3 = Chr(-32419) _
                Or s3 = Chr(-32443) _
                Or s3 = Chr(-32420) _
                Or s3 = Chr(-32421) _
                Or s3 = Chr(-31585) _
                Or s3 = Chr(-31574) _
                Or s3 = Chr(-32443) _
                Or s3 = Chr(45) _
                Or s3 = Chr(176)) Then
                c.Value = Left$(s, i) & Chr(-32388) & _
                Mid$(s, i + 2, Len(s) - i)
            End If
        Next i
    Next c
    MsgBox "done"
End Sub

Private Sub Worksheet_Change(ByVal Target As Range)'住所自動入力
'郵便番号辞書を有効にしておく
    Dim objApp As Application
    If Target.Columns.Count > 1 _
    Or Target.Rows.Count > 1 Then Exit Sub
    '値が変更されたのがa列の場合
    If (Target.Column = 1) Then
        '入力された値の長さが3以上の場合
        'かつ、B列の同じ行のセルが空白の場合
        If (Len(Target.Value) >= 3) And _
        (Cells(Target.Row, 2) = "") Then
             Set objApp = Application
             objApp.EnableEvents = False

             Cells(Target.Row, 2).Value = _
                 StrConv(Target.Value, vbWide)
             Cells(Target.Row, 2).Select
             'F2 => Shift+Home => F13
             SendKeys "{F2}", True
             SendKeys "+{HOME}", True
             SendKeys "{F13}", True

             objApp.EnableEvents = True
        End If
    End If
End Sub

Dim flg As Boolean 'Trueの場合のみ入力可
Const myMsg As String = """2005/5/24"" 或いは" & vbNewLine _
                & " ""5/24"" のように入力してください"
Private Sub TextBox1_Change()
    With TextBox1
        If flg = False Then
            MsgBox myMsg, vbInformation _
                , "半角数字と""/""のみ入力可"
                flg = True
                Me.TextBox1.Value = Empty
                Exit Sub
        Else
            '2005/5…の段階で日付と判断させない
            If InStr(Left$(.Value, 4), "/") = 0 Then
                Exit Sub
            
            ElseIf IsDate(.Value) Then
               .Value = Format$(.Value, "yyyy/m/d")
            End If
        End If
    End With
End Sub

    As MSForms.ReturnInteger, ByVal Shift As Integer)
    'ATOKユーザーだと意味無いけど気休めで

    If Not (KeyCode >= 48 And KeyCode <= 57 Or KeyCode = 191) Then

End Sub

        Next ii

            MsgBox myMsg, vbInformation, """日""まで入力しろボケがっ！"
            .Value = Empty
        End If
    End With
End Sub
