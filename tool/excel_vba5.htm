Sub smp()
    '参照設定でscripting runtimeにチェック
    Dim WB As Workbook
    Dim myPt As String
    Dim myBkname As String
    Dim myFldrName As String
    Dim FSO As New FileSystemObject
    Dim i As Long
    i = 0
    myPt = ThisWorkbook.Path
    myBkname = ThisWorkbook.FullName
    myFldrName = myPt & "\BunkatuFiles"
    '同名のフォルダ・ファイルが存在しても
    '上書きしないように
    'フォルダを作成してその中にファイルを作成
    Do While FSO.FolderExists(myFldrName)
        i = i + 1
        myFldrName = _
            Left$(myFldrName, Len(myPt) + 13) _
            & "(" & i & ")"
    Loop
    FSO.CreateFolder myFldrName
    With FSO
        .CopyFile _
            Source:=myBkname, _
            Destination:=myFldrName & "\分割A.xls"
        .CopyFile _
            Source:=myBkname, _
            Destination:=myFldrName & "\分割B.xls"
    End With
    Set FSO = Nothing
    '分割したブックのマクロを実行させてみる
    With Application
        .Run "'" & myFldrName _
         & "\分割A.xls" & "'!sampMacro"
        Workbooks("分割A.xls").Close _
         SaveChanges:=False
        .Run "'" & myFldrName _
        & "\分割B.xls" & "'!sampMacro"
        Workbooks("分割B.xls").Close _
         SaveChanges:=False
    End With
End Sub

        StartTime = GetTickCount

                MsgBox "大当たり!！"

                MsgBox "小当たり！"

            ElseIf [a1].Value = 3 Then

Function GetWeekNum(myDay As Date) As Integer
    Dim intYobi As Integer
    Dim i As Integer
    Dim intDay As Integer
    i = 1
    'パラメータが元旦から数えて何日目か
    intDay = _
        CLng(myDay) - _
        CLng(DateSerial(Year(myDay), 1, 1) - 1)
    
    '年の初回の日曜日は何日目か
        intYobi = Int(DateSerial(Year(myDay), 1, i) Mod 7)
    Do While intYobi <> 1
        '土曜～日曜：0～6
        i = i + 1
        intYobi = Int(DateSerial(Year(myDay), 1, i) Mod 7)
    Loop
    If i = 1 Then
        GetWeekNum = intDay \ 7
    Else
        GetWeekNum = (intDay + 14 - i) \ 7

    End If
End Function

Option Explicit

Sub Zen2Han_Main()
    'テキストファイルを取得
    On Error GoTo ERR_Zen2Han
    Dim TgtTxtFile As String
    TgtTxtFile = _
        Application.GetOpenFilename _
            ("テキストファイル(*.txt),*.txt")
    'キャンセルされたら処理を抜ける
    If StrConv(TgtTxtFile, vbUpperCase) = _
        "FALSE" Then Exit Sub
    Application.ScreenUpdating = False
    s_Readtxtfile TgtTxtFile
    s_Zen2Han
    s_MakeTxt
    s_Writetxtfile
    Application.ScreenUpdating = True
    MsgBox strOutFileName & " を作成しました"
    Exit Sub
ERR_Zen2Han:
    MsgBox _
        Err.Number & " : " & Err.Description
End Sub
Sub s_Readtxtfile(myFile As String)
    'テキストファイルを読み込む
    myFnum = FreeFile
    i = 0
    Open myFile For Input As #myFnum
    Do Until EOF(myFnum)
        Line Input #myFnum, myBuf
        i = i + 1

		'エクセルワークシートでは行頭に"'"があった場合
		'値として評価されなくなってしまうので、結果
		'行頭にある"'"は読み込むことは出来ても、出力できない。
		'この現象を回避するために、"'" & myBuf として
		'inputしておく。

        Sheet1.Cells(i, 1).Value = "'" & myBuf
    Loop
    Close #myFnum
End Sub

Sub s_Zen2Han()
    '全角→半角
    Dim myRng As Range
    Dim myCell As Range
    Dim myPrgf As String
    Dim myChr As String
    Set myRng = Sheet1.UsedRange
    For Each myCell In myRng
        myPrgf = myCell.Value
        For i = 1 To Len(myPrgf)
            myChr = Mid(myPrgf, i, 1)
            'カタカナはそのまま、ハイフンや _
            '長音記号もそのまま
            If Not (Asc(myChr) >= -31936 _
            And Asc(myChr) <= -31850) _
            And Not (Asc(myChr) >= 166 _
            And (Asc(myChr) <= 223)) _
            And Not _
                        Application.WorksheetFunction.Or( _
                     Chr(-32419) = myChr _
                    , Chr(-32443) = myChr _
                    , Chr(-32420) = myChr _
                    , Chr(-32421) = myChr _
                    , Chr(-31585) = myChr _
                    , Chr(-31574) = myChr _
                    , Chr(-32443) = myChr _
                    , Chr(45) = myChr _
                    , Chr(176) = myChr) Then
                myPrgf = Replace(myPrgf, myChr, _
                    StrConv(myChr, vbNarrow))
                myCell.Value = myPrgf
            End If
        Next i
    Next myCell
End Sub
Sub s_MakeTxt()
    'txtファイルを作成します
    i = 0
    myDir = ThisWorkbook.Path & "\"
    strOutFileName = myDir & conOutFileName
    Do Until Len(Dir(strOutFileName & ".txt")) = 0
        i = i + 1
        strOutFileName = myDir & conOutFileName _
                                 & "(" & i & ")"
    Loop
    strOutFileName = strOutFileName & ".txt"
    myID = Shell("COMMAND.COM /C DIR>" & strOutFileName)
    Application.Wait Now() + TimeValue("00:00:01")
    '作成されるまで待つ
End Sub

Sub s_Writetxtfile()
    '変換後ファイルを出力します
    '書き出し
    myFnum = FreeFile
    Open strOutFileName For Output As #myFnum
    For i = 1 To Sheet1.UsedRange.Rows.Count
        Print #myFnum, Sheet1.Cells(i, 1)
    Next i
    Close #myFnum
    Sheet1.UsedRange.Clear
End Sub

Function GetAddressFromZip(ByVal myZip As String) As String
    Dim myFixChimei(1 To 5) As String * 512
    Dim myChimei(1 To 5) As String
    Dim i As Long

    Application.Volatile
    myZip = StrConv(myZip, vbNarrow)
    With CreateObject("VBScript.RegExp")
        .Pattern = "\D"
        .Global = True
        myZip = .Replace(myZip, vbNullString)
    End With

    myZip = Left(myZip & String(7, "0"), 7)
    myZip = Format(myZip, "000-0000")
    fnStartYubin7
    GetZipDecision myZip, _
        myFixChimei(1), myFixChimei(2), myFixChimei(3), _
        myFixChimei(4), myFixChimei(5)
    For i = 1 To 5
        myChimei(i) = Replace(myFixChimei(i), _
                        vbNullChar, vbNullString)
        myChimei(i) = Replace(myChimei(i), _
                        Space(1), vbNullString)
    Next
    GetAddressFromZip = Join(myChimei, vbNullString)
End Function

    rngTemp.ClearContents
    rngTemp.Copy rngTemp.EntireColumn
    '特定のセルの Width を元に，複写先の ColumnWidth を変更．
    FitColumn rngTemp, rngTarget
    rngTemp.EntireRow.AutoFit
    'オートフィット後の高さ　の取得
    rngTarget.Copy rngTemp
    rngTemp.UnMerge
    rngTemp.EntireRow.AutoFit
    '複写先の Height を元に，結合されている範囲の Heightを変更．
    FitHeight rngTarget, rngTemp
End Sub

Private Sub FitColumn(ByVal 対象範囲 As Range, ByVal 目標Width As Range)
    Dim 比率 As Double ' Width / ColumnWidth
    Dim 差分 As Double ' 対象範囲.Width - 目標Width.Width
    Dim Counter As Long '“差分”が収束しないときのガス抜き用
    Dim aColumn As Range
    Dim 幅 As Double '目標Width の ColumnWidth の和
    For Each aColumn In 目標Width.Columns
        幅 = 幅 + aColumn.ColumnWidth
    Next
    対象範囲.EntireColumn.ColumnWidth = 幅
    比率 = 対象範囲.Width / 対象範囲.ColumnWidth
    For Counter = 0 To 9
        差分 = (対象範囲.Width - 目標Width.Width) / 比率
        If Abs(差分) < 0.25 / 比率 Then Exit For
        対象範囲.EntireColumn.ColumnWidth = 対象範囲.EntireColumn.ColumnWidth - 差分
    Next
    Debug.Print 対象範囲.Width, 目標Width.Width, Counter
End Sub

Private Sub FitHeight(ByVal 対象範囲 As Range, ByVal 目標Height As Range)
    Dim Height1 As Double, Height2 As Double
    Dim 差分1 As Double, 差分2 As Double ' 対象範囲.Height - 目標Height.Height
    Height1 = 目標Height.RowHeight / 対象範囲.Rows.Count
    対象範囲.EntireRow.RowHeight = Height1
    Height1 = 対象範囲.EntireRow.RowHeight
    差分1 = 対象範囲.Height - 目標Height.Height
    Height2 = Height1 - Sgn(差分1) * 0.25
    対象範囲.EntireRow.RowHeight = Height2
    差分2 = 対象範囲.Height - 目標Height.Height
    If Abs(差分1) < Abs(差分2) Then
        対象範囲.EntireRow.RowHeight = Height1
    End If
    Debug.Print 対象範囲.Height, 目標Height.Height
End Sub

Private Sub Worksheet_Activate()
    myDrct = Application.MoveAfterReturnDirection
    Application.MoveAfterReturnDirection = xlDown
    With Me
        Set myR = .Range("a:a,c:c")
        .Unprotect
        .Cells.Locked = True
        myR.Locked = False
        .Protect
    End With
End Sub

Private Sub Worksheet_Deactivate()
    On Error Resume Next
    Application.MoveAfterReturnDirection = myDrct
    On Error GoTo 0
End Sub

Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Application.MoveAfterReturnDirection = xlDown
    With Me
        If Target.Rows.Count > 1 Then Exit Sub
        If Target.Columns.Count > 1 Then Exit Sub
        If Intersect(Target, myR) Is Nothing Then
            On Error Resume Next
            Application.EnableEvents = False
            Target.Offset(-1).Next.Select
            Application.EnableEvents = True
            If Err.Number <> 0 Then
                Target.Next.Select
            End If
            On Error GoTo 0
        Else
            Exit Sub
        End If
    End With
End Sub
