'テキストボックスの日付入力規則
Dim flg As Boolean 'Trueの場合のみ入力可
Const myMsg As String = """2005/5/24"" 或いは" & vbNewLine _
                & " ""5/24"" のように入力してください"
Private Sub TextBox1_Change()
    With TextBox1
        If flg = False Then
            MsgBox myMsg, vbInformation _
                , "半角数字と""/""のみ入力可"
                flg = True
                .Value = Empty
                Exit Sub
        Else
            '2005/5…の段階で日付と判断させない
            If InStr(Left$(.Value, 4), "/") = 0 Then
                Exit Sub
            
            ElseIf IsDate(.Value) Then
               .Value = Format$(.Value, "yyyy/m/d")
            End If
        End If
    End With
End Sub

Private Sub TextBox1_KeyDown(ByVal KeyCode _
    As MSForms.ReturnInteger, ByVal Shift As Integer)
    'ATOKユーザーだと意味無いけど気休めで
    '日本語入力OFF
    IMEMode = wdIMEModeOff
    flg = True
    'BSとDELとEnterは操作可
    If KeyCode = 8 Or KeyCode = 46 Or KeyCode = 13 Then Exit Sub
    If Not (KeyCode >= 48 And KeyCode <= 57 Or KeyCode = 191) Then
       flg = False
    End If
End Sub

Private Sub TextBox1_LostFocus()
    '2005/5 でも日付として解釈されるので
    '日にちまで入れない場合はリセット
    Dim i As Long, ii As Long
    i = 0
    With Me.TextBox1
        For ii = 1 To Len(.Value)
            If Mid(.Value, ii, 1) = "/" Then
                i = i + 1
            End If
        Next ii

        If i < 2 Then
            MsgBox myMsg, vbInformation, """日""まで入力！"
            .Value = Empty
        ElseIf Not IsDate(.Value) Then
            MsgBox "そのような日にちはありません"
            flg = True
            .Value = Empty
        End If
    End With
End Sub

Sub フォルダを指定して画像を取り込む()

            End If
        Next myfile
    End With
End Sub

Sub アクティブな文書中の表に同じスタイルを設定する()
    Dim myDoc As Document
    
    Set myDoc = ActiveDocument
    
    For Each Table In myDoc.Tables
        With Table
            .AutoFormat Format:=wdTableFormatClassic4, ApplyBorders _
            :=True, ApplyShading:=True, ApplyFont:=True, ApplyColor:=True, _
            ApplyHeadingRows:=True, ApplyLastRow:=False, ApplyFirstColumn:=True, _
            ApplyLastColumn:=False, AutoFit:=True
            .Rows.Alignment = wdAlignRowCenter
        End With
    Next Table
End Sub

'ヘッダーのテキストおｗ削除してから挿入
Private Sub InsertHeader()

    Const cnsUketuke As String = "受付番号:"
    
    With ThisDocument
        .Windows(1).View.SeekView = _
            wdSeekCurrentPageHeader
        
        'Application.Selection
        With .Parent.Selection
            '常に上書き
            .HomeKey Unit:=wdLine, Extend:=wdMove
            .EndKey Unit:=wdLine, Extend:=wdExtend
            .Delete
            
            .TypeText Text:=cnsUketuke & file_Num
            .ParagraphFormat.Alignment = _
                wdAlignParagraphRight
        End With
        .Windows(1).View.SeekView = _
            wdSeekMainDocument
    End With
End Sub

'挿入するテキストの作成
Private Function file_Num() As String
    Dim FSO As Object
    Set FSO = VBA.CreateObject("Scripting.FileSystemObject")
    
    Dim ext As String
    Dim meName As String
    Dim rslt As String
    
    meName = ThisDocument.FullName
    '拡張子
    ext = FSO.GetExtensionName(meName)
    meName = VBA.StrConv(meName, vbNarrow)
    rslt = Right$(meName, Len(meName) - InStrRev(meName, ")"))
    rslt = Left(rslt, Len(rslt) - Len(ext) - 1)
    file_Num = rslt
    
End Function

Sub 選択範囲の拡大_全角()

　Dim myRange As Range
　
　Set myRange = Selection.Range
　
　'先頭を拡張
　Do While Not myRange.Characters.First.Previous Like "[a-zA-Z0-9 ]"
　　myRange.Start = myRange.Start - 1
　Loop
　
　'末尾を拡張
　Do While Not myRange.Characters.Last.Next Like "[a-zA-Z0-9 ]"
　　myRange.End = myRange.End + 1
　Loop
　
　'Rangeオブジェクトを選択
　myRange.Select
　
　'Rangeオブジェクトを解放
　Set myRange = Nothing
　
End Sub

Public Sub TypeTextFont(ByVal Text As String, ByVal FontName As String)
  Dim r As Word.Range
  
  Set r = Selection.Range
  r.Text = Text
  r.Select
  With Application.CommandBars.FindControl(ID:=1728)
    If .Enabled = True Then .Text = FontName 'フォント(ComboBox)を操作
  End With
  Selection.Collapse wdCollapseEnd
End Sub

Sub スタイルを全てコピーするマクロ()

　Dim myFilePath As String　' Wordファイルパス
　Dim myFD As FileDialog
　
　Set myFD = Application.FileDialog(msoFileDialogFilePicker)

　'------------------------------------------------
　'Wordファイルの選択
　'------------------------------------------------
　With myFD
　
　　.AllowMultiSelect = False
　　.Title = "Wordファイルを選択してください"
　
　　With .Filters
　　　.Clear
　　　.Add "すべてのWordファイル", "*.doc; *.docx"
　　End With
　
　　If .Show = -1 Then
　　　myFilePath = .SelectedItems(1)
　　　.Filters.Clear
　　Else
　　　.Filters.Clear
　　　Exit Sub
　　End If
　
　End With

　'------------------------------------------------
　'スタイルをコピーする
　'------------------------------------------------
　If Dir(myFilePath, vbNormal) <> "" Then
　　ActiveDocument.CopyStylesFromTemplate Template:=myFilePath
　End If
　
　Set myFD = Nothing
　
End Sub
