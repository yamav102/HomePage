'シリアル値を渡して年度年を返します
Function GetNendoYear(argDay As Date) As Long
    Dim nen As Long
    nen = VBA.DateTime.Year(argDay)
    
    '1,2,3月の年度は年から1を引く
    If VBA.DateTime.Month(argDay) < 4 Then
        GetNendoYear = nen - 1
    Else
        GetNendoYear = nen
    End If
End Function

Function lastRw(ws As Worksheet, clm As Long) As Long
    Dim rw As Long
    With ws
        rw = _
            .Cells(.Rows.Count, clm).End(xlUp).Row
    End With
    If rw = DataWsRw.titleRw Then
        Err.Raise 65535, , msgRwlimit, ThisWorkbook.Name
    End If
    lastRw = rw
End Function
