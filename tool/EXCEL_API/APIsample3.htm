テンポラリファイルを作成する

Const MAX_PATH = 260     'パスの最大長

'---------------------------------------------------------------------
'テンポラリファイル用のディレクトリのパス名を取得する関数
'
'戻り値     成功 = lpBufferバッファに書き込まれた文字数(終端のNullを除く)
'           失敗 = 0
'---------------------------------------------------------------------
Declare Function GetTempPath Lib "kernel32" _
    Alias "GetTempPathA" _
    (ByVal nBufferLength As Long, _
    ByVal lpBuffer As String) As Long

'-----------------------------------------------------------------------
'テンポラリファイルのファイル名もしくは実際にテンポラリファイルを作成する関数
'
'戻り値     成功 = テンポラリファイル名に使われた一意な数
'           失敗 = 0
'-----------------------------------------------------------------------
Declare Function GetTempFileName Lib "kernel32" _
    Alias "GetTempFileNameA" _
    (ByVal lpszPath As String, _
    ByVal lpPrefixString As String, _
    ByVal wUnique As Long, _
    ByVal lpTempFileName As String) As Long
'summary:テンポラリファイルを作成する
Sub GetTempFileName_Sample()

    Dim strBuffer As String * MAX_PATH          'テンポラリディレクトリのパス
    Dim strPath As String                       'テンポラリディレクトリのパス
    
    Dim strPrefixString As String               'テンポラリファイル名のプリフィックス
    Dim strTempFileName As String * MAX_PATH    'テンポラリファイル名
    
    Dim strTempFN As String                     'テンポラリファイル名
    Dim rc As Long
    
    'テンポラリファイル用のディレクトリのパスを取得
    rc = GetTempPath(Len(strBuffer), _
                    strBuffer)
                    
    '取得したパスから必要な文字列だけを抽出
    strPath = Left(strBuffer, InStr(strBuffer, vbNullChar) - 1)
    
    'ファイル名のプリフィックスを指定３文字を超える設定は桁落ちします
    strPrefixString = "api"
    
    'テンポラリファイルを作成
	'第1引数にカレントディレクトリを指定するときは"."を代入する
	'第3引数に0&を指定する事でユニークなファイル名を作成する
    rc = GetTempFileName(strPath, _
                        strPrefixString, _
                        0&, _
                        strTempFileName)
                        
    '取得したファイル名から必要な文字列だけを抽出
    strTempFN = Left(strTempFileName, _
        InStr(strTempFileName, vbNullChar) - 1)
        
    MsgBox strTempFN & " を作成しました" & Chr(13) & Chr(13) & _
        "確認後自分で削除してください"
    
    Debug.Print strTempFN
End Sub

ファイルを検索する

Const MAX_PATH = 260     'パスの最大長

'--------------------------------------------
'指定されたファイルを検索する関数
'
'戻り値     成功 = バッファにコピーされた文字数
'           失敗 = 0
'--------------------------------------------
Declare Function SearchPath Lib "kernel32" _
    Alias "SearchPathA" _
    (ByVal lpPath As String, _
    ByVal lpFileName As String, _
    ByVal lpExtension As String, _
    ByVal nBufferLength As Long, _
    ByVal lpBuffer As String, _
    ByVal lpFilePart As String) As Long

'**********************************
'summary:ファイルを検索する
'**********************************
'検索フォルダにNULLを指定すると複数のフォルダを検索します。フォルダの順序については
'MSDNを参照して下さい。サブフォルダまで検索することはありません。
Sub SearchPath_Sample()
    Dim strPath As String           '検索するディレクトリ名
    Dim strFileName As String       '検索対象のファイル名(フルパスも可能)
    Dim strExtention As String      '拡張子
    Dim lngBufferLength As Long     'パス名を受け取るバッファのサイズ
    Dim strBuffer As String * MAX_PATH 'パス名を受け取るバッファ
    Dim strFilePart As String       'ディレクトリ名なしのファイル名へのポインタ
    Dim rc As Long
    
    'ファイルを検索するパスを指定
    strPath = vbNullString
    
    'ファイル名を指定
    strFileName = "Notepad"
    
    '拡張子を指定
    strExtention = ".exe"
    
    
    'パス名を受け取るバッファのサイズを指定
    lngBufferLength = MAX_PATH
    
    'ファイルを検索
    rc = SearchPath(strPath, _
                    strFileName, _
                    strExtention, _
                    lngBufferLength, _
                    strBuffer, _
                    strFilePart)
    
    '取得したパス名とファイル名を表示
    MsgBox Left(strBuffer, InStr(strBuffer, vbNullChar) - 1)
End Sub

拡張子に関連付けられたプログラム名を取得する

Const MAX_PATH = 260     'パスの最大長

'-------------------------------------------------------------------
'指定された拡張子に関連付けられているプログラム名とハンドルを取得する関数
'
'戻り値     成功 = 32を超える数値
'           失敗 = 31以下の数値
'-------------------------------------------------------------------
Declare Function FindExecutable Lib "shell32.dll" _
    Alias "FindExecutableA" _
    (ByVal lpFile As String, _
    ByVal lpDirectory As String, _
    ByVal lpResult As String) As Long


Public Const ERROR_FILE_NOT_FOUND = 2&  'ファイルが見つからない
Public Const ERROR_PATH_NOT_FOUND = 3&  'パスが見つからない
Public Const ERROR_BAD_FORMAT = 11&     'ファイルが無効
'summary:拡張子に関連付けられたプログラム名を取得する
Sub FindExecutable_Sample()
    
    Dim strFile As String               '指定の拡張子を持つファイル名
    Dim strDirectory As String          'デフォルトのディレクトリ名
    Dim strResult As String * MAX_PATH  'プログラム名を受け取るバッファ
    Dim rc As Long
    
    Dim intRtn As Integer
    
    intRtn = _
        MsgBox("このサンプルを実行するためには C:\Windows\setuplog.txt がなければなりません" _
        & Chr(13) & Chr(13) & _
        "準備は整っていますか?", vbYesNo + vbQuestion)
        
    If intRtn = vbNo Then Exit Sub
    
    '拡張子(ファイル名)を指定
    strFile = "setuplog.txt"
    
    'ディレクトリを指定
    '<使用しているOSに応じてディレクトリ名を変更してください>
    strDirectory = "C:\Windows"
    
    '関連付けられているプログラム名とハンドルを取得
    rc = FindExecutable(strFile, _
                        strDirectory, _
                        strResult)
    
    If rc > 32 Then                 '関数呼び出し成功
        MsgBox "拡張子 txt に関連付けられたファイル: " & _
            Left(strResult, InStr(strResult, vbNullChar) - 1)
    Else                            '関数呼び出し失敗
        Select Case rc
            Case 0&
                MsgBox "メモリ不足です"
            Case ERROR_FILE_NOT_FOUND
                MsgBox "ファイルが見つかりません"
            Case ERROR_PATH_NOT_FOUND
                MsgBox "パスが見つかりません"
            Case ERROR_BAD_FORMAT
                MsgBox "ファイルが無効です"
            Case 31&
                MsgBox "関連付けられているプログラムがありません"
        End Select
    End If
    
End Sub

Sub SHGetSpecialFolderLocation_Sample()
    Dim lngWndOwner As Long
    Dim lngFolder As Long
    Dim lngPidl As Long
    Dim strPath As String * MAX_PATH
    Dim rc As Long
    
    Dim strClassName As String  'クラス名
    Dim strPrompt As String
     
    'Excelのクラス名を指定
    strClassName = "XLMAIN"

    'ダイアログボックスの親ウィンドウのハンドル
    lngWndOwner = FindWindow(strClassName, _
                            Application.Caption)
    'Excel2002の場合は　変数=Application.Hwnd　でもよい
    
    'インターネットキャッシュディレクトリを指定
    lngFolder = CSIDL_INTERNET_CACHE
    
    'インターネットキャッシュディレクトリのpidlを取得
    rc = SHGetSpecialFolderLocation(lngWndOwner, _
                                    lngFolder, _
                                    lngPidl)
                                    
    '関数の呼び出しが成功したら
    If lngPidl <> 0 Then
    
        'SHGetSpecialFolderLocationで取得した値からフォルダのフルパスを取得
        rc = SHGetPathFromIDList(lngPidl, _
                                strPath)

        strPrompt = Left(strPath, InStr(strPath, vbNullChar) - 1)
        MsgBox "インターネットキャッシュディレクトリ: " & strPrompt
    Else
        MsgBox "関数の呼び出しに失敗しました"
    End If
    
    '割り当てられたメモリを開放
    Call CoTaskMemFree(lngPidl)
End Sub

ドライブの種類を判別する

'-----------------------------------
'指定されたドライブの種類を調べる関数
'
'戻り値     ドライブの種類を識別する値
'------------------------------------
Declare Function GetDriveType Lib "kernel32" _
    Alias "GetDriveTypeA" _
    (ByVal nDrive As String) As Long

'GetDriveType関数の返り値
 Const DRIVE_UNKNOWN = 0&     'ドライブの種類が判別できない
 Const DRIVE_NO_ROOT_DIR = 1& '指定のルートディレクトリが存在しない
 Const DRIVE_REMOVABLE = 2&   'ドライブからディスクを抜くことができる
 Const DRIVE_FIXED = 3&       'ドライブからディスクを抜くことができない
 Const DRIVE_REMOTE = 4&      'リモート(ネットワーク)ドライブ
 Const DRIVE_CDROM = 5&       'CD-ROMドライブ
 Const DRIVE_RAMDISK = 6&     'RAMディスク
'summary:ドライブの種類を判別する
Sub GetDriveType_Sample()
    Dim strDrive As String
    
    strDrive = InputBox(Prompt:="ドライブ名を入力してください", _
        Default:="C:\")

    If strDrive <> "" Then
        Select Case GetDriveType(strDrive)
            Case DRIVE_UNKNOWN
                MsgBox "ドライブの種類が判別できません"
            Case DRIVE_NO_ROOT_DIR
                MsgBox "指定のルートディレクトリが存在しません"
            Case DRIVE_REMOVABLE
                MsgBox "リムーバルディスクです"
            Case DRIVE_FIXED
                MsgBox "固定ディスクです"
            Case DRIVE_REMOTE
                MsgBox "リモート(ネットワーク)ドライブです"
            Case DRIVE_CDROM
                MsgBox "CD-ROMドライブです"
            Case DRIVE_RAMDISK
                MsgBox "RAMディスクです"
        End Select
    End If
End Sub

エラー処理

Win32APIのエラーを取得する GetLastError 関数は、VBA から使用できないという制限がある。 この仕様制限に対するため LastDLLError プロパティが用意されている。 

'-------------------------------------------------------
'エラーの処理モードを設定する関数
'
'戻り値     以前のエラーモード
'-------------------------------------------------------
Declare Function SetErrorMode Lib "kernel32" _
    (ByVal wMode As Long) As Long


'OSは致命的なエラーに関するメッセージボックスを表示せず
'呼び出し側のプロセスにエラーを送る
 Const SEM_FAILCRITICALERRORS = &H1& 
'OSは一般保護違反メッセージボックスを表示しない。
'独自に一般保護違反を処理するデバッグアプリケーションにおいてのみ使用することと
 Const SEM_NOGPFAULTERRORBOX = &H2&
'WindowsNT RISCのみ : メモリアラインメントの違反を自動的に修復する。
'x86プロセッサでは効果がない
 Const SEM_NOALIGNMENTFAULTEXCEPT = &H4&
'OSはファイルが見つからなかったときのメッセージボックスを表示せず、
'呼び出し側プロセスにエラーを送る
 Const SEM_NOOPENFILEERRORBOX = &H8000&
'-------------------------------------------------------
'ファイルやディレクトリの属性を返す関数
'属性を取得するVBAのGetAttr関数を使うことが推奨されます。
'
'戻り値     成功 = 指定したファイルまたはディレクトリの属性
'           失敗 = &HFFFFFFFF(-1)
'-------------------------------------------------------
Declare Function GetFileAttributes Lib "kernel32" _
    Alias "GetFileAttributesA" _
    (ByVal lpFileName As String) As Long

'読み取り専用
 Const FILE_ATTRIBUTE_READONLY = &H1&
'隠しファイルまたは隠しディレクトリ
 Const FILE_ATTRIBUTE_HIDDEN = &H2&
'OSのファイルの一部。またはOS専用のファイル
 Const FILE_ATTRIBUTE_SYSTEM = &H4&
'ディレクトリ
 Const FILE_ATTRIBUTE_DIRECTORY = &H10&
'アーカイブファイルまたはアーカイブディレクトリ
 Const FILE_ATTRIBUTE_ARCHIVE = &H20&
'特に属性はない
 Const FILE_ATTRIBUTE_NORMAL = &H80&
'テンポラリファイル
 Const FILE_ATTRIBUTE_TEMPORARY = &H100&
'圧縮ファイルまたは圧縮ディレクトリ
 Const FILE_ATTRIBUTE_COMPRESSED = &H800&

'ドライブの準備ができていない
 Const ERROR_NOT_READY = 21&
'ディスクがフォーマットされていない
 Const ERROR_GEN_FAILURE = 31&

'summary:ファイルの属性を取得する（エラー処理　その1）
Sub GetFileAttributes_Sample()
    Dim strDrive As String
    Dim rc As Long
    
    Dim intRtn As Integer

    intRtn = MsgBox("AドライブがFDでないときにはプログラムを書き換えてください" _
        & Chr(13) & Chr(13) & _
        "AドライブはFDですか?", vbYesNo + vbQuestion)

    If intRtn = vbNo Then Exit Sub
    
    'ルートディレクトリを指定
    strDrive = "A:\"
    
    'ドライブの種類を取得
    rc = GetDriveType(strDrive)
    
    'ドライブがFDの場合は
    If rc = DRIVE_REMOVABLE Then
    
        'OSによるエラー処理を抑止して呼び出し側のプロセスにエラーを送る
        rc = SetErrorMode(SEM_FAILCRITICALERRORS)
        
        'ルートディレクトリのファイル属性を取得
        rc = GetFileAttributes(strDrive)
        
        'エラーが発生
        If rc = &HFFFFFFFF Then
            Select Case Err.LastDllError
                Case ERROR_NOT_READY
                    MsgBox "デバイスの準備ができていません"
                Case ERROR_GEN_FAILURE
                    MsgBox "ディスクがフォーマットされていません"
                Case Else
                    MsgBox "状態を取得できません"
            End Select
        Else
            MsgBox rc
        End If
        
        'エラーの処理モードを既定に設定
        rc = SetErrorMode(0&)
    End If
End Sub

'-------------------------------------------------------
'エラー番号をメッセージに変換する関数
'
'戻り値     成功 = バッファに格納されたバイト数(終端のNULLを文字除く)
'           失敗 = 0
'-------------------------------------------------------
Declare Function FormatMessage Lib "kernel32" _
    Alias "FormatMessageA" _
    (ByVal dwFlags As Long, _
    lpSource As Any, _
    ByVal dwMessageId As Long, _
    ByVal dwLanguageId As Long, _
    ByVal lpBuffer As String, _
    ByVal nSize As Long, _
    Arguments As Long) As Long


'FormatMessage関数にバッファの割り当てを要求する。
'このフラグを指定した場合、lpBufferパラメータにPVOID型変数へのポインタを、
'nSizeパラメータに出力メッセージバッファに割り当てるバイト数(ANSI版)
'または文字数(Unicode版)の最小値を指定する。FormatMessage関数は、
'書式化済みのメッセージを格納しておくのに十分なサイズのバッファを自動的に割り当て、
'そのバッファへのポインタをlpBufferポインタに格納する。
'このバッファが不要になったら、LocalFree関数を使ってバッファを解放する。
'FORMAT_MESSAGE_ALLOCATE_BUFFERフラグを指定しなかった場合は、

'summary:ファイルの属性を取得する（エラー処理　その2）
Sub GetFileAttributes_Sample2()
    Dim strDrive As String
    Dim strBuffer As String * 1024
    Dim rc As Long
    
    Dim intRtn As Integer

    intRtn = MsgBox("Aドライブがフロッピーデバイスでないときにはプログラムを書き換えてください" _
        & Chr(13) & Chr(13) & _
        "Aドライブはフロッピーデバイスですか?", vbYesNo + vbQuestion)

    If intRtn = vbNo Then Exit Sub
        
    'ルートディレクトリを指定
    strDrive = "A:\"
    
    'ドライブの種類を取得
    rc = GetDriveType(strDrive)
    
    'ドライブがリムーバルディスクのある場合は
    If rc = DRIVE_REMOVABLE Then
    
        'OSによるエラー処理を抑止して呼び出し側のプロセスにエラーを送る
        rc = SetErrorMode(SEM_FAILCRITICALERRORS)
        
        'ルートディレクトリのファイル属性を取得
        rc = GetFileAttributes(strDrive)
        
        'エラーが発生
        If rc = &HFFFFFFFF Then
            rc = FormatMessage(FORMAT_MESSAGE_IGNORE_INSERTS Or _
                                FORMAT_MESSAGE_FROM_SYSTEM, _
                                ByVal vbNullString, _
                                Err.LastDllError, _
                                0&, _
                                strBuffer, _
                                Len(strBuffer), _
                                0&)
            MsgBox _
            Left(strBuffer, InStr(strBuffer, vbNullChar) - 1)
        End If
        
        'エラーの処理モードを既定に設定
        rc = SetErrorMode(0&)
    End If
End Sub
