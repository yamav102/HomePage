Sub SetWindowLong_Sample()
    Dim hwnd As Long
    Dim lngNewLong As Long
    Dim rc As Long
    
    Dim strClassName As String  'クラス名
    
    MsgBox _
        "タイトルバーの［コントロールメニュー］ボックス、" & _
        "［閉じる］［最大化］［最小化］ボタンを無効にします" & vbCr & _
        "有効にするときには「SetWindowLong_Sample2」を実行してください"

    'Excelのクラス名を指定
    strClassName = "XLMAIN"
    
    'ウィンドウのハンドルを取得
    hwnd = FindWindow(strClassName, _
                    Application.Caption)
    'Excel2002の場合は　変数=Application.Hwnd　でもよい
    
    'ウィンドウに関する情報を取得
    lngNewLong = GetWindowLong(hwnd, _
                            GWL_STYLE)
    
    'ウィンドウの属性を変更
    rc = SetWindowLong(hwnd, _
                    GWL_STYLE, _
                    lngNewLong And (Not WS_SYSMENU))
    
    'ウィンドウのメニューバーを再描画
    rc = DrawMenuBar(hwnd)
End Sub

定数値内容
WS_BORDER&H800000境界を持つウィンドウを作成する
WS_CAPTION&HC00000タイトルバーを持つウィンドウを作成する（暗黙にWS_BORDER スタイルを持つ）。WS_DLGFRAMEスタイルと一緒に使うことはできない
WS_CHILD&H40000000子ウィンドウを作成する。WS_POPUP スタイルと一緒に使うことはできない
WS_CLIPCHILDREN&H2000000親ウィンドウの内部で描画するときに子ウィンドウが占める領域を除外する。親ウィンドウを作成するときに使う
WS_CLIPSIBLINGS&H4000000互いに関連する兄弟ウィンドウをクリップする。つまり、ある特定の兄弟ウィンドウが描画メッセージを受け取ったときに、WS_CLIPSIBLINGS スタイルは更新される兄弟ウィンドウの領域からオーバラップするほかのすべての兄弟ウィンドウをクリップする（WS_CLIPSIBLINGS を指定しないで兄弟ウィンドウがオーバラップしていると、隣の兄弟ウィンドウのクライアント領域内を描画できるようになる）。必ずWS_CHILD スタイルと一緒に使う
WS_DISABLED&H8000000初期状態で使用禁止のウィンドウを作成する
WS_DLGFRAME&H400000二重境界を持ちタイトルを持たないウィンドウを作成する
WS_GROUP&H20000ユーザーが方向キーを使って次のコントロールに移動できるコントロールグループの最初のコントロールを指定する。最初のコントロールの後でWS_GROUP スタイルをFALSE と指定して定義したコントロールはすべて同じグループに属す。次のWS_GROUP スタイルを持つコントロールは次のグループを開始する（つまり、次のグループの始まりが前のグループの終わりになる）
WS_HSCROLL&H100000水平スクロールバーを持つウィンドウを作成する
WS_MAXIMIZE&H1000000最大表示されたウィンドウを作成する
WS_MAXIMIZEBOX&H10000最大表示ボタンを持つウィンドウを作成する
WS_MINIMIZE&H2000000初期状態でアイコン化されたウィンドウを作成する。必ずWS_OVERLAPPED スタイルと一緒に使う
WS_MINIMIZEBOX&H20000アイコン化ボタンを持つウィンドウを作成する
WS_OVERLAPPED
WS_OVERLAPPEDWINDOW		&H0&オーバラップウィンドウを作成する。オーバラップウィンドウは通常キャプションと境界を持つ WS_OVERLAPPED 、WS_CAPTION 、 WS_SYSMENU 、WS_THICKFRAME 、 WS_MINIMIZEBOX、WS_MAXIMIZEBOX スタイルを持つオーバラップウィンドウを作成する
WS_POPUP&H80000000ポップアップウィンドウを作成する。WS_CHILD スタイルと一緒に使うことはできない
WS_POPUPWINDOW WS_BORDER、WS_POPUP、WS_SYSMENU スタイルを持つポップアップウィンドウを作成する。コントロールメニューを可視にするにはWS_CAPTION スタイルとWS_POPUPWINDOWを組み合わせなければならない
WS_SYSMENU&H80000タイトルバーにコントロールメニューボックスを持つウィンドウを作成する。タイトルバーを持つウィンドウにだけ使う
WS_TABSTOP&H10000ユーザーがTab キーを使って移動できるコントロールの1 つであることを指定する。ユーザーがTab キーを押すと次にWS_TABSTOP スタイルを持つコントロールに移動できる
WS_THICKFRAME&H40000ウィンドウのサイズ変更に使うことができる、太い枠を持つウィンドウを作成する
WS_VISIBLE&H10000000初期状態で可視のウィンドウを作成するWS_VSCROLL &H200000 垂直スクロールバーを持つウィンドウを作成する

定数値内容
WS_EX_ACCEPTFILES&H10&このスタイルを使って作成されたウィンドウは、ドラッグ＆ドロップされたファイルを受け入れることを指定する
WS_EX_CLIENTEDGE&H200&ウィンドウの境界の縁を3D で見えるように指定する
WS_EX_CONTEXTHELP&H400&ウィンドウのタイトルバーに疑問符(?)を付加する。疑問符をクリックしたとき、カーソルのポインタは疑問符に変わる。子ウィンドウがクリックされたとき、子ウィンドウはWM_HELP メッセージを受け取る
WS_EX_CONTROLPARENT&H10000&Tab キーによるウィンドウ間の移動を許す
WS_EX_DLGMODALFRAME&H1&引数dwStyle にWS_CAPTION スタイルフラグを指定することによりタイトルバーを持ち（オプション）、二重境界を持つウィンドウを指定する
WS_EX_LEFT&H0&ウィンドウに一般的な左揃えプロパティを与える。これはデフォルトとなる
WS_EX_LEFTSCROLLBAR&H4000&垂直スクロールバーをクライアント領域の左に置く
WS_EX_LTRREADING&H0&左から右へのリーディングオーダープロパティを使ったウィンドウテキストを表示する。デフォルト値
WS_EX_MDICHILD&H40&MDI 子ウィンドウを作成する
WS_EX_NOPARENTNOTIFY&H4&このスタイルを指定して作成された子ウィンドウは、子ウィンドウが作成または破棄されたときに親ウィンドウにWM_PARENTNOTIFY メッセージを送らないことを指定する
WS_EX_OVERLAPPEDWINDOW WS_EX_CLIENTEDGE とWS_EX_WINDOWEDGE スタイル の組み合わせ
WS_EX_PALETTEWINDOW WS_EX_WINDOWEDGE 、WS_EX_TOOLWINDOW とWS_EX_TOPMOST スタイルの組み合わせ
WS_EX_RIGHT&H1000&ウィンドウに一般的な右揃えプロパティを与える。これはウィンドウクラスに依存する
WS_EX_RIGHTSCROLLBAR&H0&垂直スクロールバーがあるならばクライアント領域の右に置く。デフォルト値
WS_EX_RTLREADING&H2000&右から左へのリーディングオーダープロパティを使ったウィンドウテキストを表示する
WS_EX_STATICEDGE&H20000&ユーザーの入力を受け付けないアイテムのために使用する3D 境界を持つウィンドウを作成する
WS_EX_TOOLWINDOW&H80&フローティングツールバーを使用するためにツールウィンドウを作成する。ツールウィンドウは正常なタイトルバーよりも短いタイトルバーを持ち、そしてウィンドウタイトルがより小さいフォントを使っている。タスクバーまたは［Alt］＋［Tab］キーが押されたとき現れるウィンドウにはツールウィンドウは現れない
WS_EX_TOPMOST&H8&このスタイルを使って作成されたウィンドウは、すべてのトップレベルでないウィンドウの上に置かれ、そのウィンドウがアクティブでなくなってもそのままトップレベルに残ることを指定する。アプリケーションはSetWindowPos メンバ関数を使ってこの属性の追加や削除ができる
WS_EX_TRANSPARENT&H20&このスタイルを使って作成されたウィンドウは透過状態になることを指定する。つまり、このウィンドウの下にあるウィンドウが見えなくなることはない。このスタイルを使って作成されたウィンドウは、そのウィンドウの下にある兄弟ウィンドウがすべて更新されたあとにだけWM_PAINT メッセージを受け取る
WS_EX_WINDOWEDGE&H100&浮き出した境界線を持つウィンドウを指定する

レジストリ・INI ファイルの操作

レジストリは、Windows と、その上で動く高度なアプリケーションの情報を管理するための膨大なデータベースで、 次のようにカテゴリ分けされている。

▼ルートキーのカテゴリーとその内容
カテゴリー（定数）値内容
HKEY_CLASSES_ROOT&H80000000ソフトウェアの設定情報
HKEY_CURRENT_USER&H80000001特定のユーザー環境の設定情報
HKEY_LOCAL_MACHINE&H80000002ローカルコンピュータの設定情報
HKEY_USERS&H80000003すべてのユーザー環境の設定情報
HKEY_CURRENT_CONFIG&H80000005現在のハードウェアの設定情報

●VBAで出来るレジストリ操作

VBAでもレジストリを操作する事ができます。ただし、 HKEY_CURRENT_USERの下にあるSoftware内のVB and VBA Program Setting内に限られます。

'summary:●レジストリへの登録
'HKEY_CURRENT_USER\Software\VB and VBA Program Setting\に
'AppName:=MyApp, Section:=Startup, Key:=Top, Setting:=75と
'AppName:=MyApp, Section:=Startup, Key:=Left, Setting:=25
'を作成します。
Sub VBA_SaveSetting_Sample()
    SaveSetting "MyApp", "Startup", "Top", 75
    SaveSetting "MyApp", "Startup", "Left", 25
End Sub

'summary:●レジストリデータの取得
Sub VBA_GetSetting_Sample()
    Dim myRegVal As String
    
    myRegVal = GetSetting("MyApp", "Startup", "Top")
    MsgBox myRegVal    '75が返る
End Sub
'summary:●レジストリMyApp\Startup内の全てのデータを取得する
Sub VBA_GetAllSettings()
    Dim myRegvalAry As Variant
    Dim i As Integer
    
    myRegvalAry = VBA.Interaction.GetAllSettings("MyApp", "Startup")
    
    For i = 0 To UBound(myRegvalAry)
        'Left : 25、Top : 75 が返る
        Debug.Print myRegvalAry(i, 0) & " : " & myRegvalAry(i, 1)
    Next i
        
End Sub
'summary:●レジストリデータを削除する
Sub VBA_DeleteSetting()
    DeleteSetting "MyApp"
    '個別に削除することも出来ます。
    'DeleteSetting "MyApp", "Startup", "Left"
End Sub

Sub S_RegSetValueEx_Sample(phkResult As Long, DataValue As Variant)

    Dim strBuffer As String
    Dim lngBuffer As Long
    Dim bytBuffer() As Byte
    Dim rc As Long

    'データのタイプ別にレジストリに保存する
    Select Case VarType(DataValue)
        Case vbString                             '文字列
            strBuffer = DataValue
            rc = RegSetValueEx(phkResult, _
                                "String", _
                                0&, _
                                REG_SZ, _
                                ByVal strBuffer, _
                                Len(strBuffer) + 1)

        Case vbInteger, vbLong, vbBoolean   '32ビット数値
            lngBuffer = DataValue
            rc = RegSetValueEx(phkResult, _
                                "Long", _
                                0&, _
                                REG_DWORD, _
                                lngBuffer, _
                                4)

        Case vbByte Or vbArray              'バイナリデータ
            bytBuffer = DataValue
            rc = RegSetValueEx(phkResult, _
                                "Binary", _
                                0&, _
                                REG_BINARY, _
                                bytBuffer(1), _
                                5)
    End Select

    If rc <> ERROR_SUCCESS Then
        MsgBox "関数の呼び出しに失敗しました"
        End
    End If
End Sub

Sub RegEnumKeyEx_Sample()

    Const lngRegSize As Long = 256

    Dim lngKey As Long
    Dim strSubkey As String
    Dim lngResult As Long
    Dim strRegKey As String
    Dim lngIndex As Long
    Dim udtFileTime As FILETIME
    Dim astrFuncResult() As String
    Dim i As Long
    
    
    lngKey = HKEY_CURRENT_USER
    strSubkey = "Software\Microsoft\Windows\CurrentVersion"
    
    ' レジストリを開く
    If RegOpenKeyEx(lngKey _
                    , strSubkey, _
                    0&, _
                    KEY_ENUMERATE_SUB_KEYS, _
                    lngResult) = 0& Then
                    
        lngIndex = 0
        
        Do
            ' バッファの初期化
            strRegKey = String(lngRegSize, vbNull)
            ' サブキーの取得
            If RegEnumKeyEx(lngResult, _
                            lngIndex, _
                            strRegKey, _
                            lngRegSize, _
                            0&, _
                            0&, _
                            0&, _
                            udtFileTime) <> 0& Then Exit Do
                            
            ReDim Preserve astrFuncResult(lngIndex)
            astrFuncResult(lngIndex) = _
                   Left$(strRegKey, InStr(strRegKey, vbNullChar) - 1)
            lngIndex = lngIndex + 1
        Loop
        ' レジストリを閉じる
        RegCloseKey lngResult
    End If
    
    If UBound(astrFuncResult) > 0 Then
    
        For i = 0 To UBound(astrFuncResult)
            Debug.Print astrFuncResult(i)
        Next i
        
    End If
    
End Sub
