'summary:スクリーンセーバー実行までの待ち時間を取得する
Sub SystemParametersInfo_Sample()
    Dim lngParam As Long
    Dim rc As Long
    
    'スクリーンセーバーが有効か無効かを取得
    rc = SystemParametersInfo(SPI_GETSCREENSAVEACTIVE, _
                                0&, _
                                lngParam, _
                                0&)
                                
    'スクリーンセーバーが無効
    If lngParam = False Then
        MsgBox "スクリーンセーバーは無効です"
        Exit Sub
    End If
    
    'スクリーンセーバー実行までの待ち時間を取得
    rc = SystemParametersInfo(SPI_GETSCREENSAVETIMEOUT, _
                                0&, _
                                lngParam, _
                                0&)
    
    MsgBox "スクリーンセーバー実行までの待ち時間: " & lngParam & " 秒"
End Sub

グラフィックスモードを変更する

以下のサンプルは、画面を８００＊６００　２５６色に設定します。 ユーザー操作で色数を２５６に設定することは出来ません。

'--------------------------------------------------------
'ディスプレイのグラフィックスモードに関する情報を取得する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'--------------------------------------------------------
Declare Function EnumDisplaySettings Lib "user32" _
    Alias "EnumDisplaySettingsA" _
    (ByVal lpszDeviceName As String, _
    ByVal dwModeNum As Long, _
    lpDevMode As DEVMODE) As Long


'--------------------------------------------------------
'ディスプレイの設定を変更する
'
'戻り値     成功 = 0
'           失敗 = 0以外
'--------------------------------------------------------
Declare Function ChangeDisplaySettings Lib "user32" _
    Alias "ChangeDisplaySettingsA" _
    (lpDevMode As DEVMODE, _
    ByVal dwflags As Long) As Long


 Const CCHDEVICENAME = 32&
 Const CCHFORMNAME = 32&

 Const DISP_CHANGE_SUCCESSFUL = 0&
 Const DISP_CHANGE_RESTART = 1&

Type DEVMODE
    dmDeviceName As String * CCHDEVICENAME
    dmSpecVersion As Integer
    dmDriverVersion As Integer
    dmSize As Integer
    dmDriverExtra As Integer
    dmFields As Long
    dmOrientation As Integer
    dmPaperSize As Integer
    dmPaperLength As Integer
    dmPaperWidth As Integer
    dmScale As Integer
    dmCopies As Integer
    dmDefaultSource As Integer
    dmPrintQuality As Integer
    dmColor As Integer
    dmDuplex As Integer
    dmYResolution As Integer
    dmTTOption As Integer
    dmCollate As Integer
    dmFormName As String * CCHFORMNAME
    dmUnusedPadding As Integer
    dmBitsPerPel As Long
    dmPelsWidth As Long
    dmPelsHeight As Long
    dmDisplayFlags As Long
    dmDisplayFrequency As Long
End Type
'summary:グラフィックスモードを変更する
Sub ChangeDisplaySettings_Sample()

    Dim udtDevMode As DEVMODE
    Dim lngBitsPerPel As Long
    Dim lngPelsWidth As Long, lngPelsHeight As Long
    Dim lngDevCnt As Long
    Dim strMsg As String

    Const lngDevCntMax As Long = 148            '必要十分なデバイス数

    lngBitsPerPel = 256                         '色数
    lngPelsWidth = 800                          '横解像度
    lngPelsHeight = 600                         '縦解像度
    lngBitsPerPel = Log(lngBitsPerPel) / Log(2) '色数をビット数に変換
                                                '色数をビット数で指定した場合は不要
    
    For lngDevCnt = 0 To lngDevCntMax
        If EnumDisplaySettings(vbNullString, _
                                lngDevCnt, _
                                udtDevMode) = 1 Then
            With udtDevMode
                If .dmBitsPerPel = lngBitsPerPel _
                    And .dmPelsWidth = lngPelsWidth _
                    And .dmPelsHeight = lngPelsHeight Then Exit For
            End With
        End If
    Next

    If lngDevCnt = lngDevCntMax + 1 Then
        MsgBox "適切なデバイスが見つかりません", vbExclamation
        Exit Sub
    End If
    
    strMsg = "グラフィックスモードを" _
        & vbCrLf & 2 ^ lngBitsPerPel & "色、解像度" _
        & lngPelsWidth & "×" & lngPelsHeight _
        & vbCrLf & "に変更しました。"
    
    Select Case ChangeDisplaySettings(udtDevMode, 0)
        Case DISP_CHANGE_SUCCESSFUL
            MsgBox strMsg, vbInformation
        Case DISP_CHANGE_RESTART
            MsgBox strMsg & vbCrLf _
                & "新しいグラフィックスモードを有効にするには、" _
                & vbCrLf & "コンピュータを再起動する必要があります。" _
                , vbInformation
        Case Else
            MsgBox "グラフィックスモードを変更できませんでした。" _
                , vbExclamation
    End Select
End Sub
