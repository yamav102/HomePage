アプリケーションの重複起動を回避する

'クラス名またはキャプションからウィンドウハンドルを取得する関数
    '戻り値
    '成功：ウィンドウハンドル
    '失敗：NULL
	'lpClassName:

Declare Function FindWindow Lib "user32" _
    Alias "FindWindowA" _
    (ByVal lpClassName As String, _
    ByVal lpWindowName As String) As Long

'summary:電卓の重複起動を回避する
Sub FindWindow_Sample()

    Dim strClassName As String  'クラス名
    Dim rc As Long              'リターンコード
    
    Dim lngProcessID As Long    'Shell関数の戻り値
    
    'クラス名を指定
    strClassName = "SciCalc"
    
    '電卓のウィンドウハンドルを取得
    rc = FindWindow(strClassName, vbNullString)
    
    'ウィンドウハンドルが取得できた場合は起動しない
    '[0&]はC言語で書かれたAPI関数が
    '（符号なしの）長整数型で返すNULL値
    If rc <> 0& Then
        MsgBox "電卓はすでに起動されています"
        Exit Sub
    End If
    
    '電卓を起動する
    lngProcessID = Shell("Calc.exe", vbNormalFocus)
End Sub

クラス名の取得に関しては、GetClassName関数を参照。 Spy++のようなフリーの ツールを使用しても簡単に調べられる。

ウィンドウハンドルはxl2002以上であれば、 application.Hwnd とHwndプロパティを使っても取得することができる。xl2000以前の 下位互換の可能性がある場合はWin32APIを使わないと動作しない。

Shell 関数で起動したアプリケーションが終了するまで待機する

'既存のプロセスオブジェクトのハンドルを返す関数
'戻り値
        '成功：指定したプロセスのオープンハンドル
        '失敗：NULL
Declare Function OpenProcess Lib "kernel32" _
    (ByVal dwDesiredAccess As Long, _
    ByVal bInheritHandle As Long, _
    ByVal dwProcessId As Long) As Long


    'プロセスハンドルが使えるようにする
    Const PROCESS_QUERY_INFORMATION = &H400
    '指定したプロセスの終了を判定する
    '(終了していないときはSTILL_ACTIVEが格納される)
    Const STATUS_PENDING = &H103&
    Const STILL_ACTIVE = STATUS_PENDING




'指定されたプロセスの終了状態を返す関数
'戻り値
        '成功：0以外
        '失敗：0
Declare Function GetExitCodeProcess Lib "kernel32" _
    (ByVal hProcess As Long, _
    lpExitCode As Long) As Long


'オープンしているオブジェクトハンドルをクローズする関数
'戻り値
        '成功：0以外
        '失敗：0
Declare Function CloseHandle Lib "kernel32" _
    (ByVal hObject As Long) As Long

'summary:メモ帳が終了するまでプログラムの実行を停止する
Sub GetExitCodeProcess_Sample()
    
    Dim lngProcessID As Long    'Shell関数の戻り値
    Dim LngProcess As Long      'OpenProcess関数の戻り値
    Dim lngExitCode As Long     '終了コード
    Dim rc As Long              'リターンコード
    

    MsgBox "メモ帳が起動したのを確認したらメモ帳を終了してください"
    
    'メモ帳を起動する
    lngProcessID = Shell("Notepad.exe", vbNormalFocus)
    
    'Shell関数で起動したメモ帳の
    'プロセスオブジェクトのハンドルを取得
    LngProcess = OpenProcess(PROCESS_QUERY_INFORMATION, _
                            1, _
                            lngProcessID)
    
    'GetExitCodeProcessでプロセスの終了状態を取得し
    'プロセスが終了していない間はDoEventsでOSに制御を戻す
    
    Do
        rc = GetExitCodeProcess(LngProcess, lngExitCode)
        DoEvents
        
    Loop While lngExitCode = STILL_ACTIVE
    
    'オープンしているオブジェクトハンドルをクローズする
    rc = CloseHandle(LngProcess)
    
    MsgBox "メモ帳が終了しました"
            
End Sub

'新しいプロセスのウィンドウの表示方法を指定する構造体

    dwXCountChars As Long   'dwFlagsがSTARTF_USECOUNTCHARSでスクリーンバッファ幅
    dwYCountChars As Long   'dwFlagsがSTARTF_USECOUNTCHARSでスクリーンバッファ高
    dwFillAttribute As Long 'dwFlagsがSTARTF_USEFILLATTRIBUTEで前景・背景色指定（？）
    dwFlags As Long         'STARTUPINFOメンバーが使用されているかどうか決定する
    wShowWindow As Integer  'dwFlagsでSTARTF_USESHOWWINDOWかSW_SHOWDEFAULTでなければ無視される
    cbReserved2 As Integer  'Reserved for use by the C Run-time; must be zero.
    lpReserved2 As Long     'Reserved for use by the C Run-time; must be NULL.
    hStdInput As Long       'dwFlagsでSTARTF_USESTDHANDLES標準の入力ハンドル
                            '既定はキーボードバッファ
    hStdOutput As Long      'dwFlagsでSTARTF_USESTDHANDLES標準の出力ハンドル

    hStdError As Long       'dwFlagsでSTARTF_USESTDHANDLES標準誤差ハンドル

'summary:待機時間を指定してアプリケーションが終了するまで待機する
Sub WaitForSingleObject_Sample()
    
    Dim strApplicationName As String    '実行モジュール名
    Dim udtProcessAttributes As SECURITY_ATTRIBUTES
    Dim udtThreadAttributes As SECURITY_ATTRIBUTES
    Dim udtStartupInfo As STARTUPINFO
    Dim udtProcessInfomation As PROCESS_INFORMATION
    Dim lngMilliseconds As Long         'タイムアウト時間
    Dim rc As Long, rc2 As Long, rc3 As Long, rc4 As Long'リターンコード
    
    MsgBox "電卓を起動して5,000ミリ秒待機します"
        
    '実行モジュール名
    '<使用しているOSに応じて絶対パスを変更してください>
    strApplicationName = "C:\Windows\system32\Calc.exe"
    
    '構造体のバイト数を指定
    'LenBをユーザー定義型に使用するとメモリ内でのサイズを返します。
    '要素間の埋め込み(キーワード：パディング・アラインメント)も含まれています 
    udtProcessAttributes.nLength = LenB(udtProcessAttributes)
    udtThreadAttributes.nLength = LenB(udtThreadAttributes)
    udtStartupInfo.cb = LenB(udtStartupInfo)
    
    '新しいプロセスを作成
    rc = CreateProcess(strApplicationName, _
                     vbNullString, _
                     udtProcessAttributes, _
                     udtThreadAttributes, _
                     False, _
                     0, _
                     ByVal vbNullString, _
                     vbNullString, _
                     udtStartupInfo, _
                     udtProcessInfomation)

                     
    '待機するタイムアウト時間を指定
    lngMilliseconds = 5000
    
    '新しいプロセスがシグナル状態になるまで待機
    rc2 = WaitForSingleObject(udtProcessInfomation.hProcess, _
                           lngMilliseconds)
    '待機結果を表示
    Select Case rc2
        Case WAIT_OBJECT_0
            MsgBox "指定したアプリケーションは終了しました"
        Case WAIT_TIMEOUT
            MsgBox "指定した時間が経過しました"
        Case WAIT_FAILED
            MsgBox "関数の呼び出しに失敗しました"
    End Select

    'オープンしているスレッドハンドルをクローズする 
　　rc3 = CloseHandle(udtProcessInfomation.hThread)
    'オープンしているプロセスハンドルをクローズする 
    rc4 = CloseHandle(udtProcessInfomation.hProcess) 

End Sub

'システムメニューが操作されたとき、
そのウィンドウに送られるメッセージ
 Const WM_SYSCOMMAND = &H112
 'ウィンドウを終了する
 Const SC_CLOSE = &HF060&
 '失敗
 Const ERROR_SUCCESS = 0&
 'summary:メモ帳を終了させる
Sub SendMessage_Sample()
    
    Dim hwnd As Long         'ウィンドウハンドル
    Dim rc As Long           'リターンコード
    
    'メモ帳のウィンドウハンドルを取得
    hwnd = FindWindow("Notepad", _
                   vbNullString)
                   
    If hwnd = ERROR_SUCCESS Then
        
        MsgBox "メモ帳は起動していません", vbExclamation
        Exit Sub
        
    End If
    
    rc = SendMessage(hwnd, _
                    WM_SYSCOMMAND, _
                    SC_CLOSE, _
                    0)

End Sub

 'summary:全てのメモ帳を閉じる
Sub SendMessage_Sample2()
    
    Dim hwnd As Long         'ウィンドウハンドル
    Dim rc As Long           'リターンコード	
    
     'メモ帳のウィンドウハンドルを取得
    hwnd = FindWindow("Notepad", _
                  vbNullString)
                  
    If hwnd = ERROR_SUCCESS Then
        MsgBox "メモ帳は起動していません", vbExclamation
        Exit Sub
    End If
    
    Do While hwnd <> ERROR_SUCCESS
        
        rc = SendMessage(hwnd, _

    Loop
    
End Sub

アプリケーションを起動してユーザーが入力できる状態になるまで待機する

以下のサンプルは、Word を起動したときに、初期化が終了してユーザーが入力できる状 態になるまで待機するものです。Word はActiveX オートメーション対応のアプリケーショ ンですから、VBA から以下のように起動する手法は推奨できません。あくまでも WaitForInputIdle 関数の使用例として参考にしてください。 

'指定されたプロセスがユーザーからの
'入力を待っている状態になるまで待機する関数
'戻り値
    '成功：0
    '失敗：WAIT_TIMEOUT or -1&
Declare Function WaitForInputIdle Lib "user32" _
        (ByVal hProcess As Long, _
        ByVal dwMilliseconds As Long) As Long

'タイムアウト時間が経過して制御が返った
Const STATUS_TIMEOUT = &H102&   '2^32-1
'タイムアウト時間を無制限にする
Const WAIT_TIMEOUT = STATUS_TIMEOUT

'タイムアウト時間を無制限にする
Const INFINITE = &HFFFFFFFF
'summary:重たいアプリケーション（ここではWord）を開き、
'入力状態になるまで待機します
Sub WaitForInputIdle_Sample()

    Dim strApplicationName As String        '実行モジュール名
    Dim udtProcessAttributes As SECURITY_ATTRIBUTES
    Dim udtThreadAttributes As SECURITY_ATTRIBUTES
    Dim udtStartupInfo As STARTUPINFO
    Dim udtProcessInfomation As PROCESS_INFORMATION
    Dim lngMilliseconds As Long             'タイムアウト時間
    Dim rc As Long
    
    '実行モジュール名
    '<使用しているOfficeのバージョンに応じて絶対パスを変更してください>
    strApplicationName = _
            "C:\Program Files\Microsoft Office\Office\WINWORD.EXE"
            
    '構造体のバイト数を指定
    udtStartupInfo.cb = Len(udtStartupInfo)
    
    '新しいプロセスを作成
    rc = CreateProcess(strApplicationName, _
                      vbNullString, _
                      udtProcessAttributes, _
                      udtThreadAttributes, _
                      False, _
                      0, _
                      ByVal vbNullString, _
                      vbNullString, _
                      udtStartupInfo, _
                      udtProcessInfomation)
                      
    '待機するタイムアウト時間を無制限にする
    lngMilliseconds = INFINITE
    
    '新しいプロセスがユーザーの入力を受け付ける状態になるまで待機する
    rc = WaitForInputIdle(udtProcessInfomation.hProcess, _
                        lngMilliseconds)
                        
    '待機結果を表示
    Select Case rc
    
        Case 0&
            MsgBox "入力を開始できます"
            
        Case WAIT_TIMEOUT
            MsgBox "指定した時間が経過しました"
            
        Case -1&
            MsgBox "関数の呼び出しに失敗しました"
            
    End Select
    
End Sub

'summary:表示されているウィンドウのキャプションと
'クラス名の一覧を作成します。
Sub GetClassName_sample()

    Dim hwnd As Long
    Dim strClassName As String * 100 '固定長文字列
    Dim strCaption As String * 80
    Dim myRow As Long
    Dim WS As Worksheet
    
    hwnd = FindWindow(vbNullString, _
                    vbNullString)
                    
    Application.ScreenUpdating = False
    
    Set WS = Sheet1
    
    With WS
        .Columns("a:b").Clear
        .Cells(1).Resize(, 2).Value = Array("キャプション", "クラス名")
    End With
    
    Do
        'If文はTrueが評価されるのではなく、false(0)以外ならば実行される
        If IsWindowVisible(hwnd) Then
            GetWindowText hwnd, strCaption, Len(strCaption)
            GetClassName hwnd, strClassName, Len(strClassName)
            
           With WS.Cells(WS.Rows.Count, 1).End(xlUp).Offset(1)
           
                .Value = Left(strCaption _
                    , InStr(strCaption, vbNullChar) - 1)
                .Offset(, 1).Value = Left(strClassName _
                    , InStr(strClassName, vbNullChar) - 1)
            
           End With
           
        End If
        
        hwnd = GetNextWindow(hwnd, GW_HWDNEXT)
        
    Loop Until hwnd = GetNextWindow(hwnd, GW_HWNDLAST)
    
    WS.Columns("a:b").AutoFit
    Application.ScreenUpdating = True
    
End Sub

ウィンドウの表示位置とサイズを指定してアプリケーションを起動する

以下のサンプルは、98SE * XL2000で動作を確認する事ができましたが、 XP_pro * Xl2003ではサイズと位置の指定が出来ませんでした。 

'summary:メモ帳を指定した座標とサイズで起動します。
Sub CreateProcess_Sample()

    Dim strApplicationName As String    '実行モジュール名
    Dim udtProcessAttributes As SECURITY_ATTRIBUTES
    Dim udtThreadAttributes As SECURITY_ATTRIBUTES
    Dim udtStartupInfo As STARTUPINFO
    Dim udtProcessInfomation As PROCESS_INFORMATION
    Dim rc As Long                      'リターンコード
    
    '実行モジュール名
    '<使用しているOSに応じて絶対パスを変更してください>
    strApplicationName = "C:\WINDOWS\system32\notepad.exe"
    
    '構造体のバイト数を指定
    udtProcessAttributes.nLength = Len(udtProcessAttributes)
    udtThreadAttributes.nLength = Len(udtThreadAttributes)
    
    '新しいプロセスのメインウィンドウの表示状態を指定
    With udtStartupInfo
        
        .cb = Len(udtStartupInfo)
        .dwX = 50       '起動したアプリケーションウィンドウのX座標
        .dwY = 50       'Y座標
        .dwXSize = 400  'X方向のサイズ
        .dwYSize = 400  'Y方向のサイズ
        
        .dwFlags = _
            STARTF_USESHOWWINDOW Or _
            STARTF_USESIZE Or _
            STARTF_USEPOSITION
        
        .wShowWindow = SW_SHOWNORMAL
        
    End With
    
    '新しいプロセスを作成
    rc = CreateProcess(strApplicationName, _
                    vbNullString, _
                    udtProcessAttributes, _
                    udtThreadAttributes, _
                    False, _
                    0, _
                    ByVal vbNullString, _
                    vbNullString, _
                    udtStartupInfo, _
                    udtProcessInfomation)
                    
    '新しいプロセスの情報をイミディエイトウィンドウに表示
    With udtProcessInfomation
        
        Debug.Print "プロセスハンドル：" & Hex(.hProcess)
        Debug.Print "プロセスID：" & Hex(.dwProcessId)
        Debug.Print "スレッドハンドル：" & Hex(.hThread)
        Debug.Print "スレッドID：" & Hex(.dwThreadId)
        
    End With
    
End Sub

    (ByVal hwnd As Long)

 
'失敗
Const ERROR_SUCCESS = 0&
'summary:メモ帳を最前面に表示します
Sub SetForegroundWindow_Sample()

    Dim hwnd As Long    'ウィンドウハンドル
    Dim rc As Long      'リターンコード
    
    hwnd = FindWindow("Notepad", _
                        vbNullString)
    
    If hwnd = ERROR_SUCCESS Then
        
        MsgBox "メモ帳は起動していません", vbExclamation
        Exit Sub
        
    End If
    
    SetForegroundWindow hwnd
    
End Sub
