ディスクの空き容量を取得する

Win98以降の環境ではGetDiskFreeSpace関数は間違った値を返すので、 GetDiskFreeSpaceEx関数を使用します。引数としてULARGE_INTEGER型 が要求されますが、VBAでは対応する型がありません。そこで、32ビットづつ 前半後半に分けた構造体 ULARG_INTEGER を宣言し、シフト演算するという テクニックを使用しています。


'---------------------------------------------
'ディスクに関する情報(空き容量など)を取得する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'---------------------------------------------
Declare Function GetDiskFreeSpaceEx Lib "kernel32" _
    Alias "GetDiskFreeSpaceExA" _
    (ByVal lpDirectoryName As String, _
    lpFreeBytesAvailableToCaller As ULARGE_INTEGER, _
    lpTotalNumberOfBytes As ULARGE_INTEGER, _
    lpTotalNumberOfFreeBytes As ULARGE_INTEGER) As Long



'64ビットの構造体
Type ULARGE_INTEGER
    LowPart As Long
    HighPart As Long
End Type

'summary:ディスクの空き容量と総容量を取得する（2GB超）
Sub GetDiskFreeSpaceEx_Sample()
    Dim strDirectoryName As String                      'ディスクのルートディレクトリ
    Dim udtFreeBytesAvailableToCaller As ULARGE_INTEGER 'ディスクの空き容量
    Dim udtTotalNumberOfBytes As ULARGE_INTEGER         'ディスクの総容量
    Dim udtTotalNumberOfFreeBytes As ULARGE_INTEGER     'ディスクの空き容量
    
    Dim rc As Long
    
    Dim vntLowPart1 As Variant                          '空き容量
    Dim vntHighPart1 As Variant                         '空き容量
    Dim vntTotal1 As Variant                            '空き容量

    Dim vntLowPart2 As Variant                          '総容量
    Dim vntHighPart2 As Variant                         '総容量
    Dim vntTotal2 As Variant                            '総容量
        
    '<使用しているOSに応じて絶対パスを変更してください>
    strDirectoryName = "C:\"
    
    rc = GetDiskFreeSpaceEx(strDirectoryName, _
                            udtFreeBytesAvailableToCaller, _
                            udtTotalNumberOfBytes, _
                            udtTotalNumberOfFreeBytes)
        
    With udtTotalNumberOfFreeBytes
        vntLowPart1 = CDec(.LowPart)
        vntHighPart1 = CDec(.HighPart)
    End With
    
    '符号付き席数longの上限を超えた値は最上位ビットに
    '1がたって負数で返ってくる為、2^32を加算して
    '問題を回避する。
    If vntLowPart1 < 0 Then
        vntLowPart1 = vntLowPart1 + 2 ^ 32
    End If
    
    If vntHighPart1 < 0 Then
        vntHighPart1 = vntHighPart1 + 2 ^ 32
    End If
    
    '空き容量のシフト演算
    vntTotal1 = (vntHighPart1 * 2 ^ 32) + vntLowPart1
        
    With udtTotalNumberOfBytes
        vntLowPart2 = CDec(.LowPart)
        vntHighPart2 = CDec(.HighPart)
    End With
    
    If vntLowPart2 < 0 Then
        vntLowPart2 = vntLowPart2 + 2 ^ 32
    End If
    
    If vntHighPart2 < 0 Then
        vntHighPart2 = vntHighPart2 + 2 ^ 32
    End If
    '総容量のシフト演算
    vntTotal2 = (vntHighPart2 * 2 ^ 32) + vntLowPart2

    MsgBox _
        "空き容量: " & Format(vntTotal1, "#,### バイト") _
        & Chr(13) & _
        "総容量: " & Format(vntTotal2, "#,### バイト")
        
    Debug.Print "空き容量: " & Format(vntTotal1, "#,### バイト") & _
    "   " & Format(vntTotal1 / 1024 / 1024 / 1024, "0.000 Gバイト")
    Debug.Print "総容量: " & Format(vntTotal2, "#,### バイト") & _
    "    " & Format(vntTotal2 / 1024 / 1024 / 1024, "0.000 Gバイト")
    
End Sub

    "ファイル名の最大長: " & lngMaximumComponentLength _
    & Chr(13) & _

    & Chr(13) & _

Windowsのバージョン情報を取得する

VBAでこの関数を使う必要はありません。?Application.OperatingSystem で Windows (32-bit) NT 5.01 まで取得できます。 ここでは、ビット演算のサンプルとして掲載します。

ビット演算を行っているのは、"ビルド番号: " & CStr(.dwBuildNumber And &HFFFF&)この部分。 Win32s on Windows 3.1か Windows95/98 の場合のみ３２ビットの返り値の下位１６ビットだけ取り出さないとビルド番号をにならない という事情によります。３２ビットの下位１６ビットを取り出すために、 下位１６ビットが全て１であるFFFFと３２ビットの返り値をAND演算して上位１６ビットを 0 にするという処理をしています。 

'--------------------------------------
'Windowsのバージョン情報を取得する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'--------------------------------------
Declare Function GetVersionEx Lib "kernel32" _
    Alias "GetVersionExA" _
    (lpVersionInformation As OSVERSIONINFO) As Long


'Windowsのバージョン情報を受け取る構造体
Type OSVERSIONINFO
    dwOSVersionInfoSize As Long     '構造体のバイト数
    dwMajorVersion As Long          'メジャーバージョン番号
    dwMinorVersion As Long          'マイナーバージョン番号
    dwBuildNumber As Long           'ビルド番号
    dwPlatformId As Long            'プラットフォームID
    szCSDVersion(127) As Byte       'OSに関する付加情報
End Type

 Const VER_PLATFORM_WIN32s = 0&           'Win32s on Windows 3.1
 Const VER_PLATFORM_WIN32_WINDOWS = 1&    'Windows95/98
 Const VER_PLATFORM_WIN32_NT = 2&         'WindowsNT
'summary:Windowsのバージョン情報を取得する
Sub GetVersionEx_Sample()
    Dim udtOSVERSIONINFO As OSVERSIONINFO
    Dim rc As Long
    
    'Windowsのバージョン情報を受け取る構造体を初期化
    udtOSVERSIONINFO.dwOSVersionInfoSize = LenB(udtOSVERSIONINFO)
    
    'Windowsのバージョン情報を取得
    rc = GetVersionEx(udtOSVERSIONINFO)
    
    With udtOSVERSIONINFO
        MsgBox "バージョン番号: " & _
            .dwMajorVersion & "." & .dwMinorVersion
        
        Select Case .dwPlatformId
            Case VER_PLATFORM_WIN32s
                    MsgBox "プラットフォームID: " & _
                        "Win32s on Windows 3.1"
                        
            Case VER_PLATFORM_WIN32_WINDOWS
                If .dwMinorVersion = 0 Then
                    MsgBox _
                    "プラットフォームID: " & "Windows95" _
                    & vbCr & _
                    "ビルド番号: " & CStr(.dwBuildNumber And &HFFFF&)
                Else
                    MsgBox _
                    "プラットフォームID: " & "Windows98" _
                    & vbCr & _
                    "ビルド番号: " & CStr(.dwBuildNumber And &HFFFF&)
                End If
            
            Case VER_PLATFORM_WIN32_NT
                    MsgBox _
                    "プラットフォームID: " & "Windows NT" _
                    & vbCr & _
                    "ビルド番号: " & CStr(.dwBuildNumber)
        End Select
    End With
End Sub

コンピュータ名を取得／設定する

コンピュータ名を取得するだけなら?environ("computername")で済みます。

'--------------------------------------
'コンピューター名を取得する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'--------------------------------------
Declare Function GetComputerName Lib "kernel32" _
    Alias "GetComputerNameA" _
    (ByVal lpBuffer As String, _
    nSize As Long) As Long



'コンピューター名の最大長
Public Const MAX_COMPUTERNAME_LENGTH = 15
'summary:コンピュータ名を取得する
Sub GetComputerName_Sample()
    Const ComputerNameLength = MAX_COMPUTERNAME_LENGTH + 1
    Dim strBuffer As String * ComputerNameLength
    Dim lngSize As Long
    Dim rc As Long
    
    'コンピューター名の長さを設定
    lngSize = Len(strBuffer)
    
    'コンピューター名を取得
    rc = GetComputerName(strBuffer, _
                        lngSize)
                        
    'コンピューター名を表示
    MsgBox "コンピューター名: " & _
        Left(strBuffer, InStr(strBuffer, vbNullChar) - 1)
End Sub

CPU の数と種類を取得する

'-----------------------------------------------------
'システムに関する情報(プロセッサの種類など)を取得する関数
'
'戻り値     なし

Declare Sub GetSystemInfo Lib "kernel32" _
    (lpSystemInfo As SYSTEM_INFO)


'システムに関する情報を格納する構造体
Type SYSTEM_INFO
    dwOemID As Long                     'Windows95/98では0
    dwPageSize As Long                  'メモリページのサイズ

    lpMaximumApplicationAddress As Long '利用可能な最上位アドレス

 Const PROCESSOR_ARCHITECTURE_INTEL = 0&
 Const PROCESSOR_ARCHITECTURE_MIPS = 1&
 Const PROCESSOR_ARCHITECTURE_ALPHA = 2&

 Const PROCESSOR_PPC_604 = 604&
 Const PROCESSOR_PPC_620 = 620&
'summary:CPUの数と種類を取得する
Sub GetSystemInfo_Sample()
    Dim udtSYSTEMINFO As SYSTEM_INFO

    'システム情報を取得
    GetSystemInfo udtSYSTEMINFO
    
    'CPUの数と種類を表示
    With udtSYSTEMINFO
        MsgBox "CPUの数: " & .dwNumberOrfProcessors
        
        Select Case .dwProcessorType
            Case PROCESSOR_INTEL_386
                MsgBox "CPUの種類: INTEL_386"
            Case PROCESSOR_INTEL_486
                MsgBox "CPUの種類: INTEL_486"
            Case PROCESSOR_INTEL_PENTIUM
                MsgBox "CPUの種類: INTEL_PENTIUM"
            Case PROCESSOR_ARCHITECTURE_UNKNOWN
                MsgBox "CPUの種類: 不明"
            Case Else
                MsgBox "CPUの種類: その他のCPU"
        End Select
    End With
End Sub
