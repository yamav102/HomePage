以下のサンプルは、 rc = Shell("Notepad.exe " & strPath & "\sample.txt", vbNormalFocus) で代用できます。

'summary:パスを指定してテキストファイルを開く
'remark:ブックと同じディレクトリに
'sample.txtを作成してから実行します。
Sub ShellExecute_Sample()
    
    Dim strClassName As String  'クラス名
    Dim strPath As String       'パス名
    Dim hwnd As Long            '親ウィンドウのハンドル
    Dim strOperation As String  '実行する操作
    Dim strFile As String       'ファイル名かディレクトリ名
    Dim rc As Long              'リターンコード
    
    strPath = ActiveWorkbook.Path
    
    'クラス名を指定
    strClassName = "XLMAIN"
    
    'Excelのウィンドウハンドルを取得
    hwnd = FindWindow(strClassName, _
                    Application.Caption)
    '↑Excel2002-3であれば、次の書き方でOK
    'hwnd = Application.hwnd
    
    '実行する操作を指定
    strOperation = "open"
    
    'ファイル名を指定
    strFile = strPath & "\sample.txt"
    
    '関連付けられているプログラムでファイルを開く
    rc = ShellExecute(hwnd, _
                   strOperation, _
                   strFile, _
                   vbNullString, _
                   vbNullString, _
                   SW_SHOWNORMAL)
                   
    '関数の呼び出し失敗
    If rc <= 31 Then
        Select Case rc
            Case 0&
                MsgBox "メモリ不足です"
            Case ERROR_FILE_NOT_FOUND
                MsgBox "ファイルが見つかりません"
            Case ERROR_PATH_NOT_FOUND
                MsgBox "パスが見つかりません"
            Case ERROR_BAD_FORMAT
                MsgBox "ファイルが無効です"
            Case Else
                MsgBox "エラーが発生しました"
        End Select
    End If
                   
End Sub

'-------------------------------------------------
'拡張子に関連付けられたプログラムを実行する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'-------------------------------------------------
Declare Function ShellExecuteEx Lib "Shell32.dll" _
    Alias "ShellExecuteExA" _
    (lpExecInfo As SHELLEXECUTEINFO) As Long



'アプリケーションに関する情報を格納する構造体
Type SHELLEXECUTEINFO
    cbSize As Long          '構造体のバイト数
    fMask As Long           '構造体の有効なメンバを指定する
    hwnd As Long            '親ウィンドウのハンドル
    lpVerb As String        '実行する操作
    lpFile As String        'ファイル名もしくはディレクトリ名
    lpParameters As String  'アプリケーションに渡すパラメータ
    lpDirectory As String   '作業ディレクトリ名
    nShow As Long           '表示方法を指定するフラグ
    hInstApp As Long        '関数の実行結果を受け取るバッファ
    lpIDList As Long        '以下、オプションフィールド
    lpClass As String
    hkeyClass As Long
    dwHotKey As Long
    hIcon As Long
    hProcess As Long
End Type
'lpClassで与えるクラス名を使用する
 Const SEE_MASK_CLASSNAME = &H1&
 'hKeyClassで与えるクラスキーを使用する
 Const SEE_MASK_CLASSKEY = &H3&
 'lpIDListで与えられるitem idリストを使用する
 Const SEE_MASK_IDLIST = &H4&
 'アプリケーションを起動するためlpIDListで与えられるitem idリストを使用する
 Const SEE_MASK_INVOKEIDLIST = &HC&
 'hIconで与えられるアイコンを使用する
 Const SEE_MASK_ICON = &H10&
 'dwHotKeyで与えられるホットキーを使用する
 Const SEE_MASK_HOTKEY = &H20&
 '関数実行後プロセスの実行を継続する
 Const SEE_MASK_NOCLOSEPROCESS = &H40&
 'ドライブと接続する
 Const SEE_MASK_CONNECTNETDRV = &H80&
 '関数から戻る前に終了するためDDE接続を待つ
 Const SEE_MASK_FLAG_DDEWAIT = &H100&
 'lpDirectoryまたはlpFileで与えられる文字列内の環境変数を展開する
 Const SEE_MASK_DOENVSUBST = &H200&
 'エラー発生時にエラーメッセージボックスを表示しない
 Const SEE_MASK_FLAG_NO_UI = &H400&

'summary:[ファイルのプロパティ]ダイアログボックスを開く
Sub ShellExecuteEx_Sample()
     
     Dim udtSHELLEXEUTEINFO As SHELLEXECUTEINFO
     Dim strClassName As String     'クラス名
     Dim hwnd As Long               '親ウィンドウのハンドル
     Dim rc As Long                 'リターンコード
     
     'クラス名を指定
     strClassName = "XLMAIN"
     
     'EXCELのウィンドウハンドルを取得
     hwnd = FindWindow(strClassName, _
                    Application.Caption)
     'xl2002以降ならhwnd=Application.Hwnd
     
     With udtSHELLEXEUTEINFO
     
        'LenBをユーザー定義型に使用するとメモリ内でのサイズを返します。
        '要素間の埋め込み(キーワード：パディング・アラインメント)も含まれています
        .cbSize = LenB(udtSHELLEXEUTEINFO)
        .fMask = SEE_MASK_INVOKEIDLIST Or SEE_MASK_NOCLOSEPROCESS _
            Or SEE_MASK_FLAG_NO_UI
        .hwnd = hwnd
        .lpVerb = "properties"
        '<使用しているOSに応じて絶対パスを変更してください">
        .lpFile = "C:Windows\system32\Notepad.exe"
        .lpParameters = vbNullString
        .lpDirectory = vbNullString
        .nShow = 0&
        .hInstApp = 0&
        .lpIDList = 0&
        
     End With
     
     rc = ShellExecuteEx(udtSHELLEXEUTEINFO)
    
End Sub

フロッピーディスクをフォーマットするダイアログを表示させる

'-------------------------------------------------
'フロッピーディスクをフォーマットするダイアログを表示させる
'
'戻り値     成功 = nFormatIDの値
'           失敗 = -1
'           キャンセル = -2
'-------------------------------------------------
Declare Function SHFormatDrive Lib "Shell32.dll" _
    (ByVal hwnd As Long, _
    ByVal nDrive As Long, _
    ByVal nFormatID As Long, _
    ByVal nOption As Long) As Long

 Const SHFMT_OPT_QUICK = 0&   'クイックフォーーマット
 Const SHFMT_OPT_FULL = 1&    '通常のフォーマット
 Const SHFMT_OPT_SYSONLY = 2& '起動専用

 Const SHFMT_ERROR = -1&      'フォーマット中にエラーが発生
 Const SHFMT_CANCEL = -2&     'フォーマットが取り消しされた
 Const SHFMT_NOFORMAT = -3&   'フォーマットができないドライブを指定した
'-----------------------------------
'指定されたドライブの種類を調べる関数
'
'戻り値     ドライブの種類を識別する値
'------------------------------------
Declare Function GetDriveType Lib "kernel32" _
    Alias "GetDriveTypeA" _
    (ByVal nDrive As String) As Long

'GetDriveType関数の返り値
 Const DRIVE_UNKNOWN = 0&     'ドライブの種類が判別できない
 Const DRIVE_NO_ROOT_DIR = 1& '指定のルートディレクトリが存在しない
 Const DRIVE_REMOVABLE = 2&   'ドライブからディスクを抜くことができる
 Const DRIVE_FIXED = 3&       'ドライブからディスクを抜くことができない
 Const DRIVE_REMOTE = 4&      'リモート(ネットワーク)ドライブ
 Const DRIVE_CDROM = 5&       'CD-ROMドライブ
 Const DRIVE_RAMDISK = 6&     'RAMディスク
'summary:フロッピディスクをフォーマットするダイアログを
'表示させる
Sub SHFormatDrive_Sample()
    
    Dim hwnd As Long            '親ウィンドウのハンドル
    Dim lngDrive As Long        'ドライブ
    Dim lngFOrmatID As Long     'フォーマットの形式
    Dim lngOption As Long       'フォーマットの種類
    Dim rc As Long              'リターンコード
    
    Dim strDrive As String      'ドライブ名
    Dim strClassName As String  'クラス名
    
    strDrive = InputBox _
        (Prompt:="フロッピーのドライブ名を入力して下さい", _
        Default:="A:\")
    
    strDrive = UCase(strDrive)  '小文字を大文字に変換
    
    'ドライブの種類を取得
    rc = GetDriveType(strDrive)
    
    'フロッピーディスクでない場合
    If rc <> DRIVE_REMOVABLE Then
        MsgBox strDrive & " はフロッピーディスクではありません"
        Exit Sub
    End If
    
    strDrive = Left(strDrive, 1)
    
    'クラス名を指定
    strClassName = "XLMAIN"
    
    'Excelのウィンドウハンドルを取得
    hwnd = FindWindow(strClassName, _
                  Application.Caption)
    'xl2002以降ならばhwnd=Application.Hwndで可
    
    'フォーマットするドライブの指定
    'Aドライブ:0；
    'Bドライブ:1；
    '…
    lngDrive = Asc(strDrive) - 65
    
    'フォーマット形式を指定
    lngFOrmatID = 0         '1.44MB
    'lngFormatID=5          '720KB
    
    'フォーマットの種類
    lngOption = SHFMT_OPT_QUICK     'クイックフォーマット
    'lngOption = SHFMT_OPT_FULL     '通常のフォーマット
    'lngOption = SHFMT_OPT_SYSONLY  '起動専用クイックフォーマット
    
    rc = SHFormatDrive(hwnd, _
                    lngDrive, _
                    lngFOrmatID, _
                    lngOption)
                    
    '以上のコードはフォーマットのダイアログを表示するだけ。
    'ダイアログが実行かキャンセルされるまで本プロシージャ
    'を抜ける事はありません。
    
End Sub

'----------------------------------------------------------------------

Declare Function SHBrowseForFolder Lib "shell32.dll" _
    Alias "SHBrowseForFolderA" _

'----------------------------------------------------------------------
'SHBrowseForFolderで取得した値からフォルダのフルパスを取得する関数
'
'戻り値     成功 = NOERROR(=0)
'           失敗 = E_INVALIDARG(=&H80070057)
'----------------------------------------------------------------------
Declare Function SHGetPathFromIDList Lib "shell32.dll" _
    Alias "SHGetPathFromIDListA" _
    (ByVal pidl As Long, _
    ByVal pszPath As String) As Long
'----------------------------------------------------------------------
'OLEアロケータによって確保したメモリブロックを開放する関数
'引数：pv　解放されるべきメモリブロックへのポインタ
'戻り値     なし
'----------------------------------------------------------------------
Declare Function CoTaskMemFree Lib "OLE32.dll" _
    (ByVal pv As Long) As Long


'summary:フォルダ選択用のディレクトリツリーを表示する
Sub SHBrowseForFolder_Sample()

    Dim udtBROWSEINFO As BROWSEINFO
    Dim lngPidl As Long                 'SHBrowseForFolderの戻り値
    Dim strPath As String * MAX_PATH    'フォルダ名格納
    Dim rc As Long
    
    Dim strClassName As String          'クラス名
    
    'Excelのクラス名を指定
    strClassName = "XLMAIN"

    With udtBROWSEINFO
        'ダイアログボックスの親ウィンドウのハンドル
        .hwndOwner = FindWindow(strClassName, _
                                Application.Caption)
        'Excel2002の場合は　変数=Application.Hwnd　でもよい
                                
        'ルートフォルダをデスクトップにする
        .pidlRoot = CSIDL_DESKTOP
        
        'タイトル
        .lpszTitle = "フォルダを選択してください"
        
        'ディレクトリのみ選択可能にする
        .ulFlags = BIF_RETURNONLYFSDIRS
    End With

    '[フォルダの参照]ダイアログボックスを開く
    lngPidl = SHBrowseForFolder(udtBROWSEINFO)
    
    If lngPidl = 0& Then
        MsgBox "キャンセルが選択されました"
        Exit Sub
    End If
    
    'SHBrowseForFolderで取得した値からフォルダのフルパスを取得
    rc = SHGetPathFromIDList(lngPidl, strPath)
    
    MsgBox "選択したフォルダ: " & _
        Left(strPath, InStr(strPath, vbNullChar) - 1)
    
    '割り当てられたメモリを開放
    Call CoTaskMemFree(lngPidl)
End Sub

ファイル/フォルダの操作

以下のサンプルの動作は、リッチな進捗表示やごみ箱に入る事に拘らなければKillステートメントRmDir、またFileSystemObjectを使用することで代用できるものばかりです。

'----------------------------------------
'ファイル操作(コピー、移動、削除）を行う関数
'
'戻り値     成功 = 0
'           失敗 = 0以外
'-----------------------------------------
Declare Function SHFileOperation Lib "shell32.dll" _
    Alias "SHFileOperationA" _
    (lpFileOp As SHFILEOPSTRUCT) As Long

'動作方法を指定する構造体
Type SHFILEOPSTRUCT
    hwnd As Long                    'ダイアログボックスの親ウィンドウのハンドル
    wFunc As Long                   '操作内容を指定する
    pFrom As String                 '操作元のファイル名、ディレクトリ名
    pTo As String                   '操作先のファイル名、ディレクトリ名
    fFlags As Integer               '操作内容を指定する
    fAnyOperationsAborted As Long   '処理終了前にキャンセルしたときは「1」
    hNameMappings As Long           'ファイルネームマッピングオブジェクト
    lpszProgressTitle As String     'ダイアログボックスのキャプション
End Type

'操作内容を指定する定数
 Const FO_MOVE = &H1&         '移動
 Const FO_COPY = &H2&         'コピー
 Const FO_DELETE = &H3&       '削除
 Const FO_RENAME = &H4&       '名前を変更

'動作方法指定する定数
'複数の異なるディレクトリを指定する
 Const FOF_MULTIDESTFILES = &H1&
 '(使用不可)
 Const FOF_CONFIRMMOUSE = &H2&
 'プログレスバーを表示しない
 Const FOF_SILENT = &H4&
 '重複名を避ける
 Const FOF_RENAMEONCOLLISION = &H8&
 '確認なし
 Const FOF_NOCONFIRMATION = &H10&
 'FOF_RENAMEONCOLLISIONを設定したときhNameMappingsにファイル名を指定する
 Const FOF_WANTMAPPINGHANDLE = &H20&
 'ごみ箱へ
 Const FOF_ALLOWUNDO = &H40&
 'ワイルドカードを使うのでディレクトリを含まない
 Const FOF_FILESONLY = &H80&
 'プログレスバーは表示するがファイル名は表示しない
 Const FOF_SIMPLEPROGRESS = &H100&
 'ディレクトリ作成時に確認なし
 Const FOF_NOCONFIRMMKDIR = &H200&
 'エラーが発生してもダイアログボックスを表示しない
 Const FOF_NOERRORUI = &H400&
'summary:フォルダ単位でファイルをコピーする
'remark:C:\TestFolderを作成してその中にファイルを保存してから実行します。
'上書きするファイルがある場合、際警告を表示し,
'サイズの大きいファイルコピーにはプログレスバーを表示します。
Sub SHFileOperation_Sample()
    Dim udtSHFILEOPSTRUCT As SHFILEOPSTRUCT
    Dim rc As Long
    
    Dim strClassName As String  'Excelのクラス名を指定
    strClassName = "XLMAIN"

    With udtSHFILEOPSTRUCT
        'ダイアログボックスの親ウィンドウのハンドル
        .hwnd = FindWindow(strClassName, _
                            Application.Caption)
        'Excel2002の場合は　変数=Application.Hwnd　でもよい
                            
        '操作内容を「コピー」にする
        .wFunc = FO_COPY
        
        'コピー元のファイルを指定
        .pFrom = "C:\TestFolder\*.*"
        
        'コピー先のファイルを指定
        .pTo = "C:\TestFolder2"
    End With
    
    'ファイルのコピーを実行
    rc = SHFileOperation(udtSHFILEOPSTRUCT)
End Sub
