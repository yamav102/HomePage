関数名に「Private」が含まれないもの(GetProfileInt 関数など)は、操作できる対象がWin.ini ファイ ルに限定されます。それ以外の独自のINI ファイルを操作するときには、名前に「Private」 が含まれるもの（GetPrivateProfileInt 関数など）を使用します。

  • INIファイルに使用される文字は、大文字小文字を区別しません。
  • INIファイルのファイル名に絶対パスを含めないとWINDOWSディレクトリにあるものと看做されます。 INIファイルを自作する場合は絶対パスで指定します。
'test.ini
[Section1]
Key1=123
Key2=456
Key3="ABC"
[Section2]
Key4="DEF"
Key5="XYZ"

'-------------------------------------------------------
'INIファイルの文字列を整数として取得する関数
'
'戻り値     成功 = 指定したキーに対応する整数
'           失敗 = nDefault
'-------------------------------------------------------
Declare Function GetPrivateProfileInt Lib "kernel32" _
    Alias "GetPrivateProfileIntA" _
    (ByVal lpApplicationName As String, _
    ByVal lpKeyName As String, _
    ByVal nDefault As Long, _
    ByVal lpFileName As String) As Long


'summary:INIファイルの文字列を整数として取得する
Sub GetPrivateProfileInt_Sample()
    Dim strFileName As String
    Dim rc As Long
    
    strFileName = ActiveWorkbook.Path & "\Test.ini"
  
    rc = GetPrivateProfileInt("Section1", _
                            "Key1", _
                            999, _
                            strFileName)

    MsgBox "[Section1]Key1: " & rc
End Sub

'--------------------------------------------------------------
'INIファイルの文字列を取得する関数
'第1 引数にvbNullString を指定するとセクション名が列挙されます。また、
'第2 引数にvbNullString を指定するとキー名が列挙されます。
'戻り値     バッファに格納された文字数
'--------------------------------------------------------------
Declare Function GetPrivateProfileString Lib "kernel32" _
    Alias "GetPrivateProfileStringA" _
    (ByVal lpApplicationName As String, _
    ByVal lpKeyName As Any, _
    ByVal lpDefault As String, _
    ByVal lpReturnedString As String, _
    ByVal nSize As Long, _
    ByVal lpFileName As String) As Long


'summary:INIファイルの文字列を取得する
Sub GetPrivateProfileString_Sample()
    Dim strFileName As String
    Dim strReturnedString As String * 32767
    Dim strKey As String
    Dim i As Integer
    Dim blnFlag As Boolean
    Dim rc As Long
       
    strFileName = ActiveWorkbook.Path & "\Test.ini"

    rc = GetPrivateProfileString("Section2", _
                                "Key4", _
                                vbNullString, _
                                strReturnedString, _
                                Len(strReturnedString), _
                                strFileName)

    blnFlag = False

    'NULLが連続した場所がデータの終端
    For i = 1 To Len(strReturnedString)
        If Mid(strReturnedString, i, 1) = vbNullChar Then
            If blnFlag = True Then
                Exit For
            Else
                strKey = strKey & Mid(strReturnedString, i, 1)
                blnFlag = True
            End If
        Else
            strKey = strKey & Mid(strReturnedString, i, 1)
            blnFlag = False
        End If
    Next i
    
    strKey = Left(strKey, Len(strKey) - 1)
        
    Debug.Print strKey  'DEF が返る
    MsgBox "イミディエイトウィンドウで結果を確認してください"
End Sub

Sub WritePrivateProfileSection_Sample()
    Dim strFileName As String
    Dim strString As String
    Dim rc As Long
    
    strFileName = ThisWorkbook.Path & "\Test.ini"
    Debug.Print strFileName
    strString = "New Key1=ネットフェニックス" & vbNullChar & _
        "New Key2=代表取締役" & vbNullChar & _
        "New Key3=山田太郎" & vbNullChar & vbNullChar

    rc = WritePrivateProfileSection("Section1", _
                                    strString, _
                                    strFileName)
    If rc = 0 Then
        MsgBox "失敗"
    Else
        MsgBox "Test.ini の内容を確認してください"
    End If
End Sub

INI ファイルの文字列を変更するときには、WritePrivateProfileString 関数を使います。 第2 引数に指定したキーが存在するときにはそのキーの値を書き換え、存在しないときに はキーと値を追加します。また、第2 引数にvbNullString を指定すると、第1 引数で指定 したセクションが削除されます。 次が、そのサンプルプログラムです。

'--------------------------------------------------------------
'INIファイルの文字列を変更する関数
'
'戻り値     成功 = 0以外
'           失敗 = 0
'--------------------------------------------------------------
Declare Function WritePrivateProfileString Lib "kernel32" _
    Alias "WritePrivateProfileStringA" _
    (ByVal lpApplicationName As String, _
    ByVal lpKeyName As Any, _
    ByVal lpString As Any, _
    ByVal lpFileName As String) As Long


'summary:INIファイルの文字列を変更する
Sub WritePrivateProfileString_Sample()
    Dim strFileName As String
    Dim rc As Long
    
    strFileName = ThisWorkbook.Path & "\Test.ini"

    'Key3の値を変更する
    rc = WritePrivateProfileString("Section1", _
                                    "New Key3", _
                                    "牧野直子", _
                                    strFileName)

    'Key31を追加する
    rc = WritePrivateProfileString("Section1", _
                                    "New Key31", _
                                    "API for VBA", _
                                    strFileName)
    'Section2を削除する
    rc = WritePrivateProfileString("Section2", _
                                    vbNullString, _
                                    vbNullString, _
                                    strFileName)

    MsgBox "Test.ini の内容を確認してください"
End Sub

'--------------------------------------------------------
'Windows起動からの経過時間をミリ秒単位で取得する関数
'
'戻り値     成功 = Windows起動からの経過時間
'--------------------------------------------------------
Declare Function GetTickCount Lib "kernel32" () As Long


'summary:コードの実行速度を計測する。VBAのtimer関数でもできます。
Sub GetTickCount_Sample()
    Dim lngTimer1 As Long
    Dim lngTimer2 As Long
    
    '現在時間の取得
    lngTimer1 = GetTickCount()

    MsgBox "［OK］ボタンを押してください"
    
    '経過時間の取得	
    lngTimer2 = GetTickCount() - lngTimer1

    MsgBox "［OK］ボタンを押すまでに " & _
        lngTimer2 & " ミリ秒かかりました"
End Sub
'summary:'Windowsを起動してから経過した時間(分)を取得
Sub GetTickCount_Sample2()

    '「4,294,967,295」ミリ秒(49.7日)が計測の上限
    Dim lngTimer As Long
    Dim vntTimer As Variant
    Dim rc As Long
    Dim rc2 As Double
    
    'Windowsを起動してから経過した時間を取得
    lngTimer = GetTickCount()
    
    vntTimer = CDec(lngTimer)
    If vntTimer < 0 Then
        vntTimer = vntTimer + 2 ^ 32
    End If
    
    vntTimer = vntTimer / 1000 / 60
    vntTimer = Int(vntTimer)
    MsgBox "Windowsを起動してから経過した時間: " & _
        Format(vntTimer, "#,###分")
End Sub

'userform
Option Explicit

Dim lngtBlinkTime As Long

Private Sub UserForm_Initialize()
    'コントロールパネルで設定されているキャレットの点滅時間を取得
    lngtBlinkTime = GetCaretBlinkTime()

    lblBlinkTime.Caption = lngtBlinkTime & "ミリ秒"
    
    txtDuobleBTime.SetFocus
End Sub


Private Sub cmdDuobleBTime_Click()
    Dim rc As Long

    lblBlinkTime.Caption = lngtBlinkTime * 2 & "ミリ秒"
    
    txtDuobleBTime.SetFocus
    
    'カーソルの点滅時間を現在の倍に設定
    rc = SetCaretBlinkTime(lngtBlinkTime * 2)
End Sub


Private Sub cmdQuit_Click()
    Dim rc As Long
    
    MsgBox "カーソルの点滅時間を元の設定に戻します"
    
    'カーソルの点滅時間を現在の倍に設定
    rc = SetCaretBlinkTime(lngtBlinkTime)

    Unload Me
End Sub

Declare Function PlaySound Lib "winmm.dll" _
    Alias "PlaySoundA" _
    (ByVal lpszName As String, _
    ByVal hModule As Long, _
    ByVal dwFlags As Long) As Long



'演奏方法
'演奏終了まで戻らない
 Const SND_SYNC = &H0&
'非同期で再生
 Const SND_ASYNC = &H1&
'指定のファイルがないときデフォルトを実行しない
 Const SND_NODEFAULT = &H2&
'サウンドデータをメモリにロード
 Const SND_MEMORY = &H4&
'ループ再生
 Const SND_LOOP = &H8&
'すでに再生中のときは再生しない
 Const SND_NOSTOP = &H10&
'再生を中止する
 Const SND_PURGE = &H40&
'アプリケーション定義のサウンドを再生
 Const SND_APPLICATION = &H80&    
'サウンドドライバの使用中は再生しない
 Const SND_NOWAIT = &H2000
'lpszNameはシステムイベント名
 Const SND_ALIAS = &H10000
'lpszNameはファイル名
 Const SND_FILENAME = &H20000
'lpszNameはリソースID
 Const SND_RESOURCE = &H40004
'定義済みサウンドID
 Const SND_ALIAS_ID = &H110000

 Const SND_VALID = &H1F&
 Const SND_VALIDFLAGS = &H17201F
 Const SND_RESERVED = &HFF000000
 Const SND_TYPE_MASK = &H170007

'パスの最大長
 Const MAX_PATH = 260
'Wave音を鳴らす
Sub PlaySound_Sample()
    Dim strBuffer As String * MAX_PATH
    Dim strPath As String
    Dim rc As Long

    'Windowsディレクトリのパス名を取得
    rc = GetWindowsDirectory(strBuffer, _
                            Len(strBuffer))
    
    '取得したデータから必要な文字列だけを抽出
    strPath = Left(strBuffer, _
        InStr(strBuffer, vbNullChar) - 1)

    'Wave音を鳴らす
    rc = PlaySound _
        (strPath & "\Media\Windows XP Startup.wav", _
        0&, _
        SND_ASYNC Or SND_LOOP)
End Sub
