Sub フォルダーを指定してエクセルファイルに読み取りパスワードを掛ける()
    Dim i As Long, j As Long
    Dim fPath As String
    Dim vPW As Variant
    Dim vPW2 As Variant
    Dim mBk As Workbook, mSh As Worksheet
    Dim objShell As Object, objFolder As Object
    Dim FS As FileSearch
    j = 0
    Set objShell = CreateObject("Shell.Application")
    Set objFolder = objShell.BrowseForFolder(0, "フォルダを選ぶ", 0, "c:\\")
    If objFolder Is Nothing Then Exit Sub
    fPath = objFolder.Items().Item().Path
    Set FS = Application.FileSearch
    With FS
        .LookIn = fPath
        .Filename = "*.xls"
        .Execute
        If .FoundFiles.Count = 0 Then
            MsgBox "対象ファイルはありません"
            Exit Sub
        End If
    vPW = InputBox("設定するパスワードを入力してください")
    vPW2 = InputBox("確認の為もう一度入力してください")
    If vPW <> vPW2 Then
        MsgBox "パスワードが一致しません"
        Exit Sub
    Else
        For i = 1 To Len(vPW)
            If Asc(Mid(vPW, i, 1)) < 48 Or Asc(Mid(vPW, i, 1)) > 122 Then
                MsgBox _
                "パスワードの中に日本語Windows版以外のExcelでは使用できない文字が含まれています。パスワードをこのまま使用します。パスワードには、" _
                & "英数字(a-z、0-9)と(!、@、#、$、%、^、&、*、(、)、+、=、)を使用してください。", vbOKCancel + vbExclamation
            End If
        Next i
    
    End If
        With Application
            .ScreenUpdating = False
            .ShowWindowsInTaskbar = False
            .DisplayAlerts = False
        End With
        For i = 1 To .FoundFiles.Count
add_PW:
        
            On Error GoTo ERR_cnt
            Set mBk = Workbooks.Open _
                (Filename:=.FoundFiles(i), _
                IgnoreReadOnlyrecommended:=True, Notify:=False)
            With mBk
                .SaveAs FS.FoundFiles(i), , vPW
                .Close
            End With
        Next i
        With Application
            .ScreenUpdating = True
            .ShowWindowsInTaskbar = True
            .DisplayAlerts = True
        End With
    End With
    If j > 0 Then
        MsgBox j & "　個のファイルにパスワードを設定できませんでした"
    End If
    MsgBox FS.FoundFiles.Count - j _
        & "個のファイルにパスワード [" & vPW & "] を設定しました"
    Exit Sub
ERR_cnt:
    i = i + 1
    j = j + 1
    On Error GoTo 0
    Resume add_PW
End Sub

 【要参照】「Windows Scripting Host Object Model」'Command Promptを利用するファイル検索
Sub EnumFilePath(ByRef strTargetPath As String)
    Dim objWSHShell     As IWshRuntimeLibrary.WshShell
    Dim colProcessEnv   As IWshRuntimeLibrary.WshEnvironment
    Dim objFSO          As IWshRuntimeLibrary.FileSystemObject
    Dim objRedirectFile As IWshRuntimeLibrary.File
    Dim objRedirectTXT  As IWshRuntimeLibrary.TextStream
    Dim strRedirectPath As String
    Dim strComPrmptPath As String
    Dim strCommand      As String
    Dim lngReturn       As Long
    Dim varFileList     As Variant
    Dim varFilePath     As Variant
  
    Set objWSHShell = New IWshRuntimeLibrary.WshShell
    Set objFSO = New IWshRuntimeLibrary.FileSystemObject
    
    ' 一時的に作成する Redirect File Path 生成
    With objFSO
        strRedirectPath = _
            .GetSpecialFolder(TemporaryFolder) & "\" & .GetTempName
    End With
    
    ' Command Prompt の Path 取得
    Set colProcessEnv = objWSHShell.Environment("Process")
    strComPrmptPath = colProcessEnv.Item("COMSPEC")
    Set colProcessEnv = Nothing
    
    ' Command を生成
    
    strCommand = strComPrmptPath & " /C DIR """ & strTargetPath _
        & """ /A-D /B /S > """ & strRedirectPath & """"
     
    ' Command の実行
    lngReturn = objWSHShell.Run(strCommand, 7, True)
    Set objWSHShell = Nothing
    
    ' 出力した Text 内容取り出し
    If lngReturn = 0 Then
        Set objRedirectFile = objFSO.GetFile(strRedirectPath)
        Set objRedirectTXT = _
            objRedirectFile.OpenAsTextStream(ForReading)
    
        varFileList = Split(objRedirectTXT.ReadAll, vbCrLf)
    
        For Each varFilePath In varFileList
            'Debug.Print varFilePath
            With Sheet3
            .Range("a" & .Rows.Count) _
             .End(xlUp).Offset(1).Value = varFilePath
            End With
        Next varFilePath
    
        objRedirectTXT.Close
        objRedirectFile.Delete
    
        Set objRedirectTXT = Nothing
        Set objRedirectFile = Nothing
    End If
    Set objFSO = Nothing
End Sub

Sub ReadCSV()'最速CSV読み込み
    Dim myPath As String
    Dim N As Integer
    Dim D As String
    Dim myVals As Variant
    Dim rngDest As Range
    Dim i As Integer
    Dim timeS As Date
    Dim cnt As Long
    myPath = Application.GetOpenFilename( _
        "CSVファイル*.csv,*.csv")
    If VarType(myPath) = vbBoolean Then Exit Sub
    timeS = Now()
    N = FreeFile
    
    Open myPath For Input As #N
    Line Input #N, D
    Close #N
    
    myVals = Split(D, ",")
    
    Application.ScreenUpdating = False
    
    Set rngDest = Sheet1.Range("a1")
    Open myPath For Input As #N
    
    Do Until EOF(N)
        Line Input #N, D
        myVals = Split(D, ",")
        rngDest.Resize(, UBound(myVals) + 1).Value = myVals
        Set rngDest = rngDest.Offset(1)
        cnt = cnt + 1
        Application.StatusBar = cnt & " レコード読み込み中"
    Loop
    
    With Application
        .ScreenUpdating = True
        .StatusBar = False
    End With
    Debug.Print Format$(Now() - timeS, "hh:mm:ss")
    Close #N
    MsgBox "done"
End Sub
Private Type BITMAPINFOHEADER'ビットマップの幅高さ深さを取得

    biCompression   As Long
    biSizeImage     As Long
    biXPelsPerMeter As Long
    biYPelsPerMeter As Long

    biClrImportant  As Long

Sub Example1()
On Error GoTo Err_Handler

    Const SIZEOF_BMFHEADER = 14
    Const BF_SIG = &H4D42   'BM
    Dim bmih As BITMAPINFOHEADER
    Dim strFilePath As String
    Dim lngFileNum  As Long
    Dim fOpen       As Boolean
    Dim fSuccess    As Boolean
    Dim bfType      As Integer

    strFilePath = "C:\パス\ファイル名.bmp"

    lngFileNum = FreeFile
    GetAttr strFilePath
    Open strFilePath For Binary Access Read As #lngFileNum
    fOpen = True
    Get #lngFileNum, , bfType
    If bfType = BF_SIG Then
        Get #lngFileNum, 1 + SIZEOF_BMFHEADER, bmih
        If bmih.biSize >= Len(bmih) _
        And Not EOF(lngFileNum) Then
            fSuccess = True
        End If
    End If
    Close #lngFileNum
    fOpen = False

    If fSuccess Then
      With bmih
        Debug.Print .biWidth & "x" & Abs(.biHeight) & _
                    " (" & .biBitCount & "bit)"
      End With
    Else
        Debug.Print "ファイル形式が不明"
    End If

Exit_Proc:
    Exit Sub
Err_Handler:
    Debug.Print "エラー： " & Err.Description
    If fOpen Then Close #lngFileNum
    Resume Exit_Proc

End Sub
Sub ListModules()'モジュールをリストアップ
    Dim objComp As VBComponent
    Dim objCodeMod As CodeModule
    Dim StartLine As Long
    Dim ProcName As String
    Dim lngRow As Long
    
    lngRow = 1
    With ThisWorkbook.VBProject
        For Each objComp In .VBComponents
            Cells(lngRow, 1).Value = objComp.Name & _
                "(" & CompTypeToName(objComp) & ")"
            Set objCodeMod = .VBComponents(objComp.Name).CodeModule
            
            StartLine = objCodeMod.CountOfDeclarationLines + 1
            Do Until StartLine >= objCodeMod.CountOfLines
                ProcName = objCodeMod.ProcOfLine(StartLine, _
                vbext_pk_Proc)
                StartLine = StartLine + _
                    objCodeMod.ProcCountLines(ProcName, vbext_pk_Proc)
                Cells(lngRow, 2).Value = ProcName
                lngRow = lngRow + 1
            Loop
        Next objComp
    End With
End Sub

Function CompTypeToName(objComp As VBComponent) As String
    Select Case objComp.Type
    Case vbext_ct_ClassModule
        CompTypeToName = "Class Module"
    Case vbext_ct_Document
        CompTypeToName = "Document"
    Case vbext_ct_MSForm
        CompTypeToName = "MS Form"
    Case vbext_ct_StdModule
        CompTypeToName = "Standard Module"
    Case Else
    End Select
End Function
