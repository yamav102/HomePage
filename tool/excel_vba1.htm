Function testmsg() As String 'タイトル行でデータ型を定義
Select Case ActiveCell.Value
Case Is > 80
testmsg = "優"
Case Is > 60
testmsg = "良"
Case Is > 40
testmsg = "可"
Case Else
testmsg = "不可"
End Select
Case節の書き方
値が１，２，３のいずれかである場合　　　　　　　　　　　　　→Case １，２，３ 
値が１から5までの範囲のどれかである場合　　　　　　　　→Case 1 To 5 
値が１，２か５から１０までの範囲のどれかである場合　　→Case １，２，５ To 10 
値が１０以上の場合　　　　　　　　　　　　　　　　　　　　　　　→Case Is>=10

End Function

Sub TestMain()'シート上のhtmlリンクのページ画面をエクセルに貼り付ける
' 参照設定：Microsoft Forms 2.0 Object Library
' 参照設定：Microsoft HTML Object Library
' 参照設定：Microsoft Internet Controls

    Dim i As Range
    Dim j As Shape
    Dim wkURL As String
    Dim shtA As Worksheet

    Set shtA = ActiveSheet

    For Each i In shtA.UsedRange
        On Error Resume Next
        wkURL = ""
        wkURL = i.Hyperlinks(1).Address
        On Error GoTo 0
        If Not wkURL = "" Then
            Call prcPaste(wkURL)
        End If
    Next i

    For Each j In shtA.Shapes
        On Error Resume Next
        wkURL = ""
        wkURL = j.Hyperlink.Address
        On Error GoTo 0
        If Not wkURL = "" Then
            Call prcPaste(wkURL)
        End If
    Next j

End Sub

Private Sub prcPaste(ByVal AURL As String)

    Dim wkIE As New SHDocVw.InternetExplorer
    Dim wkDoc As MSHTML.HTMLDocument
    Dim wkHTML As String
    Dim wkDO As New MSForms.DataObject

    wkIE.Visible = True
    wkIE.Navigate AURL
    Do
        DoEvents
    Loop Until (Not wkIE.Busy) And _
               (wkIE.readyState = READYSTATE_COMPLETE)
    Set wkDoc = wkIE.document
    wkHTML = wkDoc.body.innerHTML
    wkIE.Quit
    wkDO.Clear
    wkDO.SetText wkHTML
    wkDO.PutInClipboard
    Worksheets.Add After:=Worksheets(Worksheets.Count)
    ActiveSheet.Range("A1").PasteSpecial
    wkDO.SetText ""
    wkDO.PutInClipboard

End Sub
