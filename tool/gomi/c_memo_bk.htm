ポインタ変数 = malloc(必要なメモリのバイトサイズ);
メモリを確保できなかった場合は NULL が返る。

領域が確保できたときの戻り値は void型のポインター。
代入するポインタ変数がint * だった場合、

ポインタ変数 = (int *)malloc(必要なメモリのバイトサイズ);

とキャストしてやらないと、C++コンパイラではエラーになる。

配列は要素数をプログラム中で変更できないので
malloc関数を使うと便利だが、配列で用が足りるなら
配列を使ったほうが安全とされる。

{
	int *p; /* 初期化していない状態のポインターは

	if( p == NULL )    /* 領域確保に失敗したか */

	free( p );  /* 領域解放 */
	
	return 0;
}

■構造体と並び替えのサンプル

#include <stdio.h>
/*10人分の成績を降順に並べます*/
/* result構造体 */
struct result{
	int score;		/*点数*/
	char name[20];	/*名前20文字*/
};

#define SMPLECNT 10

void show_all(struct result data[]);
void input_data(struct result data[]);

void main()
{
	int i, j, maxindex;
	struct result taihi;	
	struct result kekka[SMPLECNT];
	
	input_data(kekka);

	printf("並び替え前\n");
	
	show_all(kekka);

	for (i = 0; i < SMPLECNT - 1; i++) {

		/* i番目から順に比較し最大レコードの Index を求める */

		for (maxindex = i, j = i; j < SMPLECNT - 1; j++) {
		
			if (kekka[maxindex].score > kekka[j + 1].score) {
				maxindex = maxindex;

			}
			else{
				maxindex = j + 1;				
			}				
		}				

		/* 最大レコードを i 番目に移動 */
		
		taihi = kekka[i];
		kekka[i] = kekka[maxindex];
		kekka[maxindex] = taihi;

	}

	printf("並び替え後\n");
	
	show_all(kekka);
}
void input_data(struct result data[])
{
	/* 10人分のデータ入力 */

	int i;

	printf("%d人分のデータを入力します\n", SMPLECNT);
	for (i = 0; i < SMPLECNT; i++) {		
		printf("名前%d人目:", i + 1);
		scanf("%s", &data[i].name);
		printf("成績%d人目:", i + 1);
		scanf("%d", &data[i].score);
	}

}
void show_all(struct result data[])
{
	/* 10人分のデータを表示 */
	int i;

	for (i = 0; i < SMPLECNT; i++) {
		printf("成績：%d    \t名前：%s\n", data[i].score, data[i].name);

	}
}
