Sub TestMain()'シート上のhtmlリンクのページ画面をエクセルに貼り付ける
' 参照設定：Microsoft Forms 2.0 Object Library
' 参照設定：Microsoft HTML Object Library
' 参照設定：Microsoft Internet Controls

    Dim i As Range
    Dim j As Shape

    Dim shtA As Worksheet

    For Each i In shtA.UsedRange
        On Error Resume Next
        wkURL = ""

        On Error GoTo 0
        If Not wkURL = "" Then
            Call prcPaste(wkURL)
        End If
    Next i

    For Each j In shtA.Shapes
        On Error Resume Next
        wkURL = ""
        wkURL = j.Hyperlink.Address
        On Error GoTo 0
        If Not wkURL = "" Then
            Call prcPaste(wkURL)
        End If
    Next j

End Sub

Private Sub prcPaste(ByVal AURL As String)

    Dim wkIE As New SHDocVw.InternetExplorer
    Dim wkDoc As MSHTML.HTMLDocument
    Dim wkHTML As String
    Dim wkDO As New MSForms.DataObject

    wkIE.Visible = True
    wkIE.Navigate AURL
    Do
        DoEvents
    Loop Until (Not wkIE.Busy) And _
               (wkIE.readyState = READYSTATE_COMPLETE)
    Set wkDoc = wkIE.document
    wkHTML = wkDoc.body.innerHTML
    wkIE.Quit
    wkDO.Clear
    wkDO.SetText wkHTML
    wkDO.PutInClipboard
    Worksheets.Add After:=Worksheets(Worksheets.Count)
    ActiveSheet.Range("A1").PasteSpecial
    wkDO.SetText ""
    wkDO.PutInClipboard

End Sub

>・・・・照合の型って0じゃなくて・・・？			
大量データの検索をやる場合は、特に検査範囲が広い			
場合、Matchの照合の型0を使うのは馬鹿げたことです。			
毎回最初から一つ一つなめて検索していきますから、			
時間がかかってしょうがないです。			

Matchを使う場合は、検査範囲をソートして照合の型を
1(あるいは省略)にしたほうが良いですよ。
照合の型が0は逐次探索(馬鹿サーチ)なのにたいして、
照合の型が1や-1は二分探索(バイナリサーチ)なので、
速度が全然違います。

注意点は検査範囲はきちんとデータ範囲にすることです。
見出しを含めるのはもちろんご法度ですが、余分に
後ろの空白を含めるのもご法度です。
あべこべに照合の型0より遅くなってしまいます。

Sub 数字に続く全ての横線を置き換える()
    Dim c As Range
    Dim r As Range
    Dim s As String, s2 As String, s3 As String
    Set r = Selection
    For Each c In r
        s = c.Value
        For i = 1 To Len(s)
            s = c.Value
            s2 = Mid(s, i, 2)
            s3 = Right$(s2, 1)
            '数字の後に続く
            'ハイフン (四分)
            'ダッシュ (二分)
            'ダッシュ (全角)
            '長音記号
            '横細線素片
            '横太線素片
            'ダッシュ (二分)
            'ハイフンマイナス
            '機種依存文字の一つを"－（マイナス）"に
            '置き換えます。
            'ハイフンに置き換えるべきところですが、
            'エクセルの使用としてまま日付に変換されてしまうので
            'ハイフンを全角変換したときに"－（マイナス）"と
            '同じ文字に置き換えられるので
            'マイナス記号と置き換える事にしました。
            If IsNumeric(Left$(s2, 1)) _
                And (s3 = Chr(-32419) _
                Or s3 = Chr(-32443) _
                Or s3 = Chr(-32420) _
                Or s3 = Chr(-32421) _
                Or s3 = Chr(-31585) _
                Or s3 = Chr(-31574) _
                Or s3 = Chr(-32443) _
                Or s3 = Chr(45) _
                Or s3 = Chr(176)) Then
                c.Value = Left$(s, i) & Chr(-32388) & _
                Mid$(s, i + 2, Len(s) - i)
            End If
        Next i
    Next c
    MsgBox "done"
End Sub

Sub EnumFilePath(ByRef strTargetPath As String)
    Dim objWSHShell     As IWshRuntimeLibrary.WshShell
    Dim colProcessEnv   As IWshRuntimeLibrary.WshEnvironment
    Dim objFSO          As IWshRuntimeLibrary.FileSystemObject
    Dim objRedirectFile As IWshRuntimeLibrary.File
    Dim objRedirectTXT  As IWshRuntimeLibrary.TextStream
    Dim strRedirectPath As String
    Dim strComPrmptPath As String
    Dim strCommand      As String
    Dim lngReturn       As Long
    Dim varFileList     As Variant
    Dim varFilePath     As Variant
  
    Set objWSHShell = New IWshRuntimeLibrary.WshShell
    Set objFSO = New IWshRuntimeLibrary.FileSystemObject
    
    ' 一時的に作成する Redirect File Path 生成
    With objFSO
        strRedirectPath = _
            .GetSpecialFolder(TemporaryFolder) & "\" & .GetTempName
    End With
    
    ' Command Prompt の Path 取得
    Set colProcessEnv = objWSHShell.Environment("Process")
    strComPrmptPath = colProcessEnv.Item("COMSPEC")
    Set colProcessEnv = Nothing
    
    ' Command を生成
    
    strCommand = strComPrmptPath & " /C DIR """ & strTargetPath _
        & """ /A-D /B /S > """ & strRedirectPath & """"
     
    ' Command の実行
    lngReturn = objWSHShell.Run(strCommand, 7, True)
    Set objWSHShell = Nothing
    
    ' 出力した Text 内容取り出し
    If lngReturn = 0 Then
        Set objRedirectFile = objFSO.GetFile(strRedirectPath)
        Set objRedirectTXT = _
            objRedirectFile.OpenAsTextStream(ForReading)
    
        varFileList = Split(objRedirectTXT.ReadAll, vbCrLf)
    
        For Each varFilePath In varFileList
            'Debug.Print varFilePath
            With Sheet3
            .Range("a" & .Rows.Count) _
             .End(xlUp).Offset(1).Value = varFilePath
            End With
        Next varFilePath
    
        objRedirectTXT.Close
        objRedirectFile.Delete
    
        Set objRedirectTXT = Nothing
        Set objRedirectFile = Nothing
    End If
    Set objFSO = Nothing
End Sub

Sub Zen2Han_Main()
    'テキストファイルを取得
    On Error GoTo ERR_Zen2Han
    Dim TgtTxtFile As String
    TgtTxtFile = _
        Application.GetOpenFilename _
            ("テキストファイル(*.txt),*.txt")
    'キャンセルされたら処理を抜ける
    If StrConv(TgtTxtFile, vbUpperCase) = _
        "FALSE" Then Exit Sub
    Application.ScreenUpdating = False
    s_Readtxtfile TgtTxtFile
    s_Zen2Han
    s_MakeTxt
    s_Writetxtfile
    Application.ScreenUpdating = True
    MsgBox strOutFileName & " を作成しました"
    Exit Sub
ERR_Zen2Han:
    MsgBox _
        Err.Number & " : " & Err.Description
End Sub
Sub s_Readtxtfile(myFile As String)
    'テキストファイルを読み込む
    myFnum = FreeFile
    i = 0
    Open myFile For Input As #myFnum
    Do Until EOF(myFnum)
        Line Input #myFnum, myBuf
        i = i + 1
        Sheet1.Cells(i, 1).Value = myBuf
    Loop
    Close #myFnum
End Sub

Sub s_Zen2Han()
    '全角→半角
    Dim myRng As Range
    Dim myCell As Range
    Dim myPrgf As String
    Dim myChr As String
    Set myRng = Sheet1.UsedRange
    For Each myCell In myRng
        myPrgf = myCell.Value
        For i = 1 To Len(myPrgf)
            myChr = Mid(myPrgf, i, 1)
            'カタカナはそのまま、ハイフンや _
            '長音記号もそのまま
            If Not (Asc(myChr) >= -31936 _
            And Asc(myChr) <= -31850) _
            And Not (Asc(myChr) >= 166 _
            And (Asc(myChr) <= 223)) _
            And Not _
                        Application.WorksheetFunction.Or( _
                     Chr(-32419) = myChr _
                    , Chr(-32443) = myChr _
                    , Chr(-32420) = myChr _
                    , Chr(-32421) = myChr _
                    , Chr(-31585) = myChr _
                    , Chr(-31574) = myChr _
                    , Chr(-32443) = myChr _
                    , Chr(45) = myChr _
                    , Chr(176) = myChr) Then
                myPrgf = Replace(myPrgf, myChr, _
                    StrConv(myChr, vbNarrow))
                myCell.Value = myPrgf
            End If
        Next i
    Next myCell
End Sub
Sub s_MakeTxt()
    'txtファイルを作成します
    i = 0
    myDir = ThisWorkbook.Path & "\"
    strOutFileName = myDir & conOutFileName
    Do Until Len(Dir(strOutFileName & ".txt")) = 0
        i = i + 1
        strOutFileName = myDir & conOutFileName _
                                 & "(" & i & ")"
    Loop
    strOutFileName = strOutFileName & ".txt"
    myID = Shell("COMMAND.COM /C DIR>" & strOutFileName)
    Application.Wait Now() + TimeValue("00:00:01")
    '作成されるまで待つ
End Sub

Function GetAddressFromZip(ByVal myZip As String) As String
    Dim myFixChimei(1 To 5) As String * 512
    Dim myChimei(1 To 5) As String
    Dim i As Long

    Application.Volatile
    myZip = StrConv(myZip, vbNarrow)
    With CreateObject("VBScript.RegExp")
        .Pattern = "\D"
        .Global = True
        myZip = .Replace(myZip, vbNullString)
    End With

    myZip = Left(myZip & String(7, "0"), 7)
    myZip = Format(myZip, "000-0000")

        myFixChimei(1), myFixChimei(2), myFixChimei(3), _
        myFixChimei(4), myFixChimei(5)

                        vbNullChar, vbNullString)
        myChimei(i) = Replace(myChimei(i), _
                        Space(1), vbNullString)
    Next
    GetAddressFromZip = Join(myChimei, vbNullString)
End Function
