Sub GetRecord()


  Dim daoDB As DAO.DataBase
  Dim daoCSV As Workspace
  Dim strSQL As String
  Dim daoRs As DAO.Recordset
  Dim strPath, strFile As String
  Dim WS As Worksheet
    
  strPath = CreateObject("wscript.shell").specialfolders("desktop")  'パスのみ（ファイル名は含まない）
  strFile = "a.csv" 'strFile 'ファイル名のみ

  Set daoCSV = DBEngine.Workspaces(0)
  Set daoDB = daoCSV.OpenDatabase(strPath, 0, 0, "Text;")

  strSQL = "SELECT * FROM [" & strFile & "] where 報告日=#" & DateSerial(2004, 3, 26) & "#;"

  Set daoRs = daoDB.OpenRecordset(strSQL) ', dbOpenDynaset)

  Debug.Print daoRs.RecordCount

  Set WS = Sheet1
  
  With WS
    .UsedRange.ClearContents
    .Range("a2").CopyFromRecordset daoRs, 60
  End With

  Set daoRs = Nothing
  daoDB.Close
  Set daoDB = Nothing

End Sub

Sub Make_Tables_DATA()

    Set daoDB = CurrentDb
    
    strT = "T_Tables_Data"
    
    For Each daoT In daoDB.TableDefs
        If daoT.Name = strT Then
            DoCmd.RunSQL "Drop Table " & strT & ";"
            Exit For
        End If
    Next
    
    '▼テーブル作成
    Set daoT = daoDB.CreateTableDef(strT)
    
    '▼プロパティフィールド作成
    Set daoF = daoT.CreateField("Name", dbText)
    daoT.Fields.Append daoF
    Set daoF = daoT.CreateField("DateCreated", dbText)
    daoT.Fields.Append daoF
    Set daoF = daoT.CreateField("LastUpdated", dbText)
    daoT.Fields.Append daoF
    Set daoF = daoT.CreateField("RecordCount", dbLong)
    daoT.Fields.Append daoF

    Set daoF = daoT.CreateField("Description", dbText, 250)
    daoT.Fields.Append daoF
    
    '▼テーブル追加
    daoDB.TableDefs.Append daoT
    
    '▼プロパティ名取得のサンプル
'    Dim i
'    For i = 0 To daoDB.TableDefs.Count - 1
    '    For Each daoP In daoDB.TableDefs(i).Properties
    '        Debug.Print daoP.Name, daoP.Value
    '    Next
'    Next
    
    '▼レコードセットを開きプロパティフィールドプロパティを取り出す
    Set daoRS = daoDB.OpenRecordset(strT, dbOpenDynaset)
    
    For Each daoTable In daoDB.TableDefs
    
        daoRS.AddNew
    
        For Each daoF In daoRS.Fields
            On Error Resume Next
            If daoF.Name = "RecordCount" Then
                daoRS.Fields(daoF.Name) = f_RecCnt(daoTable.Name)
            Else
                daoRS.Fields(daoF.Name) = daoTable.Properties(daoF.Name).Value
            End If
            On Error GoTo 0
        Next daoF
    
        daoRS.Update
    
    Next daoTable
    
    'MsgBox "done", , daoDB.Name
    daoDB.Close: Set daoDB = Nothing
    
    Set daoT = Nothing

End Sub

'引数にしたいしたテーブルかクエリのレコード数を返す
Private Function f_RecCnt(ByVal T_Q_Name As String) As Long
    Dim sql As String
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    
    On Error GoTo err_handle
    
    sql = "SELECT COUNT(*) FROM " & T_Q_Name & ";"
    Set db = Application.CurrentDb
    
    Set rs = db.OpenRecordset(sql)
    f_RecCnt = rs.Fields(0).Value
    
end_handle:
    Set rs = Nothing
    Set db = Nothing
    
    Exit Function
err_handle:
    'MsgBox Err.Number & " : " & Err.Description, vbExclamation
    Resume end_handle
    
End Function

Option Compare Database
Option Explicit

    Dim xlApp As Excel.Application

Function Make_xl() As Object

    Set xlApp = New Excel.Application
    Set WB = xlApp.Workbooks.Add
    Set WS = WB.Sheets(1)
    
    xlApp.Visible = True
    
    
End Function

Sub Out_xl()
        
    Dim daoDb As DAO.Database
    Dim daoTbl As DAO.TableDef
    Dim daoRS As DAO.Recordset
    Dim strT_name As String
    Dim i As Long
    
    strT_name = "T_test"
                
    Set daoDb = CurrentDb

    End With

    Set WB = Nothing
